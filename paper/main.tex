\title{Distinct Semiprime Egyptian Fractions for Rationals with Squarefree Denominator}
\author{Yuren Tang}
\date{Major-revision manuscript workspace, 16 July 2026}

\begin{document}

\begin{abstract}
We prove that a positive rational number is a sum of reciprocals of distinct
squarefree semiprimes if and only if its reduced denominator is squarefree.
The sufficiency proof is organized around two structural prime-supply
conditions rather than around the file topology of its formalization.  The
first supplies enough primes in each sufficiently large dyadic block; the
second supplies a fixed positive reciprocal-prime mass across a long dyadic
window.  From these conditions we prove an abstract avoiding construction for
$1/b$, then obtain arbitrary numerators by disjoint iteration.

The construction combines a finite spectral selection principle with a
weighted circle method.  Control semiprimes encode residue assignments across
dyadic prime blocks; a reciprocal-mass reservoir places the Bernoulli weights
in a uniform window; and denominator-sensitive reservoir edges damp the
frequencies that are invisible to the block residues.  We isolate the three
load-bearing estimates: a global control partition obtained from level-set
counting and a localization dichotomy, a block-minor fiber estimate with exact
CRT multiplicity, and squarefree-CRT reservoir damping.  The order of the
terminal parameter choices is made explicit, including the single inequality
that makes the main Gaussian contribution dominate both minor sectors.

Finally, the two structural prime-supply conditions are derived from explicit
estimates of Rosser and Schoenfeld.  The immutable software release
\texttt{v0.0.3}, archived under DOI 10.5281/zenodo.20767390, remains the
current machine-checked evidence:
its headline theorem depends on the three standard Lean axioms and exactly two
source-transcribed Rosser--Schoenfeld inputs.  A later frozen refactor supplies
the principle/handoff architecture used for exposition here, but is not
presented as released formal authority.
\end{abstract}

\section{Result, necessity, and proof architecture}

An integer $n$ is a \emph{squarefree semiprime} if
\[
        n=pq
\]
for two distinct primes $p<q$.  Equivalently, its arithmetic functions satisfy
$\omega(n)=\Omega(n)=2$.

\begin{theorem}[Erd\H{o}s 306 in the present formulation]\label{thm:headline}
Let $q>0$ be rational.  Write $q=a/b$ in lowest terms with $a,b\in\N$ and
$b>0$.  Then $q$ is a finite sum of reciprocals of distinct squarefree
semiprimes if and only if $b$ is squarefree.
\end{theorem}

The necessary direction is immediate but important for the exact scope.  If
$S$ is a finite set of squarefree integers, then the denominator of
$\sum_{n\in S}1/n$ divides $\lcm(S)$; the latter is squarefree.  Hence every
reduced denominator arising from squarefree semiprimes is squarefree.

The sufficiency proof will be separated into four conceptual layers.

\begin{enumerate}[label=\textbf{Layer \arabic*.}]
\item \emph{Structural analytic supply.}  Dyadic blocks have enough primes for
      the control geometry and for denominator-sensitive reservoirs, while a
      long dyadic window has enough reciprocal-prime mass to create the
      required semiprime load.
\item \emph{Finite spectral construction.}  From this supply, for every
      squarefree $b\ge3$ and finite obstruction set $T$, construct an admissible
      weighted family of distinct semiprimes whose main Fourier arcs strictly
      dominate the minor arcs.
\item \emph{Exact reciprocal selection.}  A finite spectral principle converts
      the strict Fourier gap into a subset with reciprocal sum exactly $1/b$.
\item \emph{Arithmetic closure.}  Deal with $b=1,2$, iterate the avoiding
      unit-numerator statement to obtain $a/b$, and order the final finite set.
\end{enumerate}

This order is also the paper's principle/handoff order.  The structural
hypotheses are the analytic principle; the finite construction theorem is the
handoff to arithmetic; the circle method is factored into its own selection
principle and concrete certificate.

\section{The exact finite statement and avoiding reductions}

For a finite set $T\subset\N$, write
\[
 \operatorname{Rep}_T(r)
\]
when there is a finite set $S\subset\N$ such that
\begin{enumerate}[label=(\roman*)]
\item every $n\in S$ is a squarefree semiprime;
\item $S\cap T=\varnothing$;
\item $\sum_{n\in S}1/n=r$.
\end{enumerate}
The avoidance parameter is the mechanism that makes later unions disjoint.

\begin{lemma}[Unit numerator implies arbitrary numerator]\label{lem:numerator-reduction}
Assume that $\operatorname{Rep}_T(1/b)$ holds for every finite $T$.  Then for
every $a\ge1$ and finite $T$ one has $\operatorname{Rep}_T(a/b)$.
\end{lemma}

\begin{proof}
Induct on $a$.  Having obtained $S$ for $a/b$, apply the unit-numerator
hypothesis with obstruction set $T\cup S$ to obtain $U$ for $1/b$.  Then
$S\cap U=\varnothing$, and the reciprocal sums add over $S\cup U$.
\end{proof}

\begin{lemma}[The small denominators]\label{lem:small-b}
If the avoiding unit-numerator assertion holds for squarefree denominators at
least $3$, then it holds for all positive squarefree denominators.
\end{lemma}

\begin{proof}
For $b=2$, successively choose disjoint avoiding representations of $1/3$ and
$1/6$ and use $1/2=1/3+1/6$.  For $b=1$, successively choose disjoint avoiding
representations of $1/2$, $1/3$, and $1/6$ and use
$1=1/2+1/3+1/6$.
\end{proof}

Thus the analytic content is exactly the following avoiding statement for
$b\ge3$.

\section{Structural prime supply and the abstract construction theorem}

For $k\in\N$, define the dyadic prime block
\[
       \Pk_k=\{p\text{ prime}:2^k\le p<2^{k+1}\}.
\]
We isolate two conditions.

\begin{definition}[Dyadic density condition $\mathbf D$]\label{def:D}
For every $k\ge5$,
\begin{equation}\label{eq:D}
       |\Pk_k|\ge \frac{2^k}{2\log(2^k)}.
\end{equation}
\end{definition}

\begin{definition}[Reciprocal-window condition $\mathbf M$]\label{def:M}
There exists $k_1\ge5$ such that for every $k_0\ge k_1$,
\begin{equation}\label{eq:M}
  \sum_{p\in\bigcup_{k=k_0}^{3k_0}\Pk_k}\frac1p\ge\frac{21}{20}.
\end{equation}
The upper index is inclusive.  Consequently the actual prime interval is
$[2^{k_0},2^{3k_0+1})$.
\end{definition}

Condition $\mathbf D$ is a cardinality supply.  It is used to populate each
control block and to select many high reservoir primes.  Condition $\mathbf M$
is a reciprocal-mass supply.  It is used only to create a large pool of
semiprime reciprocals from which a load window can be selected.  Keeping these
roles separate prevents a common logical blur: many primes do not by themselves
provide enough reciprocal mass, while reciprocal mass alone does not provide
the uniform block cardinalities needed by the control system.

\begin{theorem}[Structural avoiding construction]\label{thm:structural-construction}
Assume $\mathbf D$ and $\mathbf M$.  Let $b\ge3$ be squarefree and let
$T\subset\N$ be finite.  Then $\operatorname{Rep}_T(1/b)$ holds.
\end{theorem}

\begin{corollary}[Structural form of Erd\H{o}s 306]\label{cor:structural-E306}
Conditions $\mathbf D$ and $\mathbf M$ imply the sufficiency direction of
Theorem~\ref{thm:headline}.
\end{corollary}

\begin{proof}
Apply Theorem~\ref{thm:structural-construction}, then
Lemmas~\ref{lem:small-b} and~\ref{lem:numerator-reduction}.
\end{proof}

Sections~\ref{sec:spectral}--\ref{sec:terminal} prove
Theorem~\ref{thm:structural-construction}.  Section~\ref{sec:RS} derives
$\mathbf D$ and $\mathbf M$ from the exact Rosser--Schoenfeld backend used by
the released formal proof.

\section{Finite spectral selection}\label{sec:spectral}

The circle method is used in a finite cyclic group, so the existence mechanism
can be stated without analytic number theory.

Let $E$ be a finite set of positive integers, let $L$ be a common multiple of
$E$, let $q_0\in\Z/L\Z$, and let $\theta_e\in[0,1]$.  Choose every $e\in E$
independently with probability $\theta_e$.  For $A\subseteq E$, set
\[
  w(A)=\prod_{e\in A}\theta_e\prod_{e\in E\setminus A}(1-\theta_e).
\]
The weighted number of subsets with
$\sum_{e\in A}L/e\equiv q_0\pmod L$ is
\[
 W_{L,q_0}(E,\theta)
 =\sum_{\substack{A\subseteq E\\
             \sum_{e\in A}L/e\equiv q_0\ (\mathrm{mod}\ L)}}w(A).
\]
Finite Fourier orthogonality gives
\begin{equation}\label{eq:finite-fourier}
 L W_{L,q_0}(E,\theta)
 =\sum_{h=0}^{L-1}
     \prod_{e\in E}\bigl((1-\theta_e)+\theta_e e^{2\pi i h/e}\bigr)
     e^{-2\pi i hq_0/L}.
\end{equation}

\begin{lemma}[Finite spectral selection]\label{lem:spectral-selection}
Partition the frequency set into disjoint sets $S_M$ and $S_m$.  If
\[
 \Re\sum_{h\in S_M}F(h)>\sum_{h\in S_m}|F(h)|
\]
and the imaginary part of the main sum is zero, where $F(h)$ is the summand in
\eqref{eq:finite-fourier}, then $W_{L,q_0}(E,\theta)>0$.  Hence an admissible
subset exists.
\end{lemma}

\begin{proof}
The real part of the full Fourier sum is at least the displayed positive
difference.  Thus the left side of \eqref{eq:finite-fourier} is positive.
All subset weights are nonnegative, so at least one admissible subset has
positive weight.
\end{proof}

For the reciprocal problem take $q_0=L/b$.  If every $e$ divides $L$ and
\begin{equation}\label{eq:no-wrap}
       \sum_{e\in E}\frac1e<1,
\end{equation}
then a congruence
$\sum_{e\in A}L/e\equiv L/b\pmod L$ is equivalent to the exact equality
$\sum_{e\in A}1/e=1/b$: both numbers lie in $[0,1)$.

It remains to build an $E$, a weight function, and main/minor estimates that
meet Lemma~\ref{lem:spectral-selection}.

\section{Dyadic control geometry}\label{sec:control}

Fix squarefree $b\ge3$ and finite $T$.  We shall choose a large bottom scale
$k_0$, a terminal scale $K$, and prime blocks
\[
       P_k\subseteq\Pk_k\qquad(k_0\le k\le K).
\]
Condition $\mathbf D$ permits all required finite selections once $k_0$ is
large.  Put
\[
       \Bset=\bigcup_{k=k_0}^{K}P_k.
\]
All primes in $\Bset$ are chosen larger than $b$ and larger than every element
of $T$.

The \emph{control graph} has vertices $\Bset$.  It includes sufficiently many
pairs inside each block and selected pairs between consecutive blocks.  The
corresponding semiprime set is
\[
       \Ectrl=\{pq:\{p,q\}\text{ is a control pair}\}.
\]
Unique factorization makes the product map injective on the ordered prime
pairs, so these denominators are distinct.

For each $p\in\Bset$, a frequency $h$ determines a residue
$a_p=h\pmod p$.  A general residue assignment is
\[
       a=(a_p)_{p\in\Bset}\in
       \mathcal A:=\prod_{p\in\Bset}\Z/p\Z.
\]
For a control pair $\{p,q\}$, let $x_{pq}(a)$ be the centered CRT
representative modulo $pq$ determined by $(a_p,a_q)$.  Define
\begin{equation}\label{eq:Qctrl}
 \Qctrl(a)=\sum_{\{p,q\}\in\mathcal C}
       \left(\frac{x_{pq}(a)}{pq}\right)^2,
 \qquad
 \sctrl^2=\sum_{\{p,q\}\in\mathcal C}\frac1{(pq)^2}.
\end{equation}
For an actual frequency $h$, the exact CRT identity is
\[
   \normdist{h/(pq)}=\frac{|x_{pq}(a(h))|}{pq}.
\]
Hence, if the final edge family contains $\Ectrl$,
\begin{equation}\label{eq:control-full-energy}
       \Qctrl(a(h))\le \QE(h)
       :=\sum_{e\in E}\normdist{h/e}^{2}.
\end{equation}

The control graph is chosen so that two quantitative facts hold for all large
$k_0$:
\begin{align}
 \sctrl&\ge \frac{1}{c_\sigma k_0 2^{k_0}},
       \label{eq:sctrl-lower}\\
 \sum_{e\in\Ectrl}\frac1e&<\frac{3}{4b}.
       \label{eq:control-load}
\end{align}
Here $c_\sigma\ge1$ is an absolute constant produced by the elementary block
calculation.  The first estimate comes from the numerous pairs in the bottom
block; the second follows because all denominators grow quadratically in the
block scale.  Neither estimate uses condition $\mathbf M$.

The associated main set in assignment space is
\begin{equation}\label{eq:assignment-main}
 \mathfrak M(C)=
 \left\{a:\exists m\in\Z,
   |m|\le \frac C{\sctrl},\quad
   a_p\equiv m\pmod p\ \text{for all }p\in\Bset\right\}.
\end{equation}
The global-control theorem bounds the Laplace mass outside this set.

\section{The global control partition}\label{sec:global}

This section isolates the most substantial combinatorial estimate.  Its input
is only the block system and its quadratic energy; the reciprocal-mass
reservoir and denominator reservoir are not involved.

\begin{theorem}[Global control partition]\label{thm:global-partition}
For every $c>0$ and every $\eta>0$, there exist a scale $k_*(c,\eta)$ and a
constant $C_{\mathrm{tail}}>0$ such that, for every admissible block system with
$k_0\ge k_*$ and every $C\ge1$,
\begin{equation}\label{eq:global-partition}
 \sum_{a\notin\mathfrak M(C)}e^{-c\Qctrl(a)}
 \le
 \frac{\eta+C_{\mathrm{tail}}e^{-cC^2/2}}{\sctrl}.
\end{equation}
\end{theorem}

We prove it through four explicit handoffs.

\subsection{Single-block rigidity and finite encoding}

For each block $P_k$, let $Q_k(a)$ be its local part of the control energy.
There are increasing forcing thresholds $R_k$ and adjacent-boundary thresholds
$\Pi_k$ with the following properties.

\begin{lemma}[Cold-block structure]\label{lem:cold-block}
For sufficiently large $k_0$, if $Q_k(a)<R_k$, then there is a dominant integer
label $\ell_k$ such that all but an explicitly energy-controlled exceptional
set of primes in $P_k$ satisfy $a_p\equiv\ell_k\pmod p$.  The number of possible
labels and exceptional sets with energy at most $R$ is bounded by an
exponential charge $e^{\varepsilon R}$ times a polynomial label window.
\end{lemma}

\begin{proof}[Mechanism]
For two residues in one block, closeness of their reciprocal phases forces a
linear congruence modulo the product of the two primes.  Centered CRT
representatives are unique inside the short interval allowed by the energy
threshold.  Thus many low-energy pair constraints decode the same integer
fingerprint.  If no label captures most residues, cross-label pairs force a
quadratic amount of energy.  Once a dominant label exists, each exception
pays a definite energy charge.  Summing the choices of exceptions uses the
weighted subset-entropy inequality
$\sum_{A}\prod_{j\in A}u_j=\prod_j(1+u_j)$, with $u_j$ chosen as an exponential
of the paid charge.
\end{proof}

\begin{lemma}[Adjacent-scale mismatch penalty]\label{lem:boundary-penalty}
If consecutive cold blocks have distinct dominant labels, then the adjacent
control pairs force energy at least $\Pi_k$ at their boundary.
\end{lemma}

\begin{proof}[Mechanism]
A residue matching $\ell_k$ in $P_k$ and a residue matching
$\ell_{k+1}$ in $P_{k+1}$ determine a centered CRT representative.  If the
labels differ, sufficiently many cross pairs have a nonzero representative of
controlled size.  Squaring and summing these contributions produces the
boundary floor.  Removing the exceptional primes costs only the already
charged exception energy.
\end{proof}

An assignment of total energy at most $R$ can now be encoded by:
\begin{enumerate}[label=(\roman*)]
\item the set $H$ of hot blocks, where $Q_k\ge R_k$;
\item the set $B$ of cold boundaries with a label mismatch;
\item dyadic energy shells for the hot and exceptional parts;
\item one label for each cold segment of blocks separated by $H\cup B$;
\item the residual fiber inside these data.
\end{enumerate}
Every hot block and every mismatch boundary consumes a definite portion of the
energy budget.  Cold labels propagate along a segment, and the fiber size is
controlled by the exception charges.

\begin{lemma}[Global level-set bound]\label{lem:level-set}
For every $0<\varepsilon<1$ there are constants $A>0$ and $k_\varepsilon$ such
that, for $k_0\ge k_\varepsilon$, $R\ge1$,
\begin{equation}\label{eq:level-set}
 \#\{a\in\mathcal A:\Qctrl(a)\le R\}
 \le e^{A J}e^{8\varepsilon R}
       \left(1+\frac{\sqrt R}{\sctrl}\right),
\end{equation}
where $J$ is the number of dyadic blocks.
\end{lemma}

\begin{proof}
Sum the encoding fibers over $H$, $B$, the shells, and segment labels.  The
weighted entropy sums for $H$ and $B$ each contribute at most
$e^{2\varepsilon R+J}$; shell counting and exception fibers contribute another
$e^{4\varepsilon R+A_0J}$.  Once a label is fixed on the first cold segment,
propagation determines all other cold labels except across charged boundaries.
The initial label has size $O(\sqrt R/\sctrl)$, because a diagonal label $m$
already contributes $m^2\sctrl^2$.  Combining the factors and enlarging the
constant of $J$ gives \eqref{eq:level-set}.
\end{proof}

\subsection{Localization outside the main assignment arc}

Let
\[
 \mathcal F_0=\min(R_{k_0},\Pi_{k_0})
\]
be the global control floor.  Define the diagonal sector outside the label
window by
\[
 \mathfrak D(C)=\left\{a:\exists m\in\Z,
  a_p\equiv m\pmod p\ \forall p,\quad
  |m|>C/\sctrl,\quad
  \Qctrl(a)=m^2\sctrl^2\right\}.
\]

\begin{lemma}[Localization dichotomy]\label{lem:localization}
For all sufficiently large $k_0$,
\[
 a\notin\mathfrak M(C)
 \quad\Longrightarrow\quad
 \Qctrl(a)\ge\mathcal F_0\quad\text{or}\quad a\in\mathfrak D(C).
\]
\end{lemma}

\begin{proof}
If a hot block exists, its local energy gives the first alternative.  If all
blocks are cold but a boundary label changes, Lemma~\ref{lem:boundary-penalty}
gives the first alternative.  Otherwise all blocks are cold, have no
exceptions after the low-energy merger, and their labels agree.  Hence every
residue is the reduction of one integer $m$.  If $|m|\le C/\sctrl$, then $a$ is
in the main arc, contrary to the hypothesis.  Otherwise the CRT energy
identity gives $\Qctrl(a)=m^2\sctrl^2$, so $a\in\mathfrak D(C)$.
\end{proof}

\subsection{High-energy absorption}

\begin{lemma}[Laplace mass above the floor]\label{lem:floor-laplace}
For every $c>0$ and $\eta>0$, after increasing $k_0$,
\begin{equation}\label{eq:floor-laplace}
 \sum_{\Qctrl(a)\ge\mathcal F_0}e^{-c\Qctrl(a)}
 \le\frac{\eta}{\sctrl}.
\end{equation}
\end{lemma}

\begin{proof}
Choose $\varepsilon$ with $8\varepsilon<c$, and put
$\bar c=(8\varepsilon+c)/2$.  Decompose assignment space into integer energy
shells and apply Lemma~\ref{lem:level-set}; the factor
$e^{8\varepsilon R}$ is summable against $e^{-\bar cR}$.  This gives a full
Laplace bound of the form
\[
 \sum_a e^{-\bar c\Qctrl(a)}
 \le e^{AJ}K_0\left(1+\frac1{\sctrl}\right).
\]
On the sector $\Qctrl\ge\mathcal F_0$, factor out
$e^{-(c-\bar c)\mathcal F_0}$.  The forcing floor grows fast enough with $k_0$
to absorb both $e^{AJ}$ and the fixed $K_0$.  Since $\sctrl\le1$ at large
scale, $1+1/\sctrl\le2/\sctrl$.  The remaining coefficient can therefore be
made at most $\eta$.
\end{proof}

\subsection{The diagonal Gaussian sector}

\begin{lemma}[Diagonal Gaussian tail]\label{lem:diagonal-tail}
For every $c>0$ there is $C_{\mathrm{tail}}>0$ such that, for all large $k_0$
and $C\ge1$,
\begin{equation}\label{eq:diagonal-tail}
 \sum_{a\in\mathfrak D(C)}e^{-c\Qctrl(a)}
 \le \frac{C_{\mathrm{tail}}e^{-cC^2/2}}{\sctrl}.
\end{equation}
\end{lemma}

\begin{proof}
A diagonal assignment is determined by its integer label, so the sum injects
into
\[
 \sum_{|m|>C/\sctrl}e^{-cm^2\sctrl^2}.
\]
Split $cm^2\sctrl^2$ into two equal halves.  The window condition extracts
$e^{-cC^2/2}$, while the remaining integer Gaussian sum is at most
$1+6/\sqrt{c\sctrl^2/2}$.  For $\sctrl\le1$ this is bounded by a constant
depending only on $c$, divided by $\sctrl$.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:global-partition}]
Apply Lemma~\ref{lem:localization} and split the off-main sum into the two
sectors.  Use Lemmas~\ref{lem:floor-laplace} and
\ref{lem:diagonal-tail}, then add the bounds.
\end{proof}

A quantifier issue is worth recording.  The entropy factor in
\eqref{eq:level-set} is $e^{AJ}$ with $A$ fixed before the final base-scale
threshold.  It is not an arbitrary constant chosen after $k_0$.  This fixed
order is what permits the control floor to absorb the entropy.

\section{Reciprocal mass, edge families, and weights}\label{sec:mass}

Condition $\mathbf M$ enters here and nowhere in the global-control proof.  Put
\[
   \Pk(k_0)=\bigcup_{k=k_0}^{3k_0}\Pk_k,
   \qquad S_1=\sum_{p\in\Pk(k_0)}\frac1p,
   \qquad S_2=\sum_{p\in\Pk(k_0)}\frac1{p^2}.
\]
For the products of two distinct primes,
\begin{equation}\label{eq:pair-mass-id}
 \sum_{\substack{p,q\in\Pk(k_0)\\p<q}}\frac1{pq}
 =\frac{S_1^2-S_2}{2}.
\end{equation}
Condition $\mathbf M$ gives $S_1\ge21/20$, while
\[
 S_2\le\sum_{n\ge2^{k_0}}\frac1{n^2}
 \le\frac1{2^{k_0}-1}.
\]
Thus, for $k_0\ge5$,
\begin{equation}\label{eq:pair-mass-half}
 \sum_{p<q}\frac1{pq}\ge\frac12.
\end{equation}

The final denominator family has three disjoint parts:
\[
       E=\Ectrl\,\dot\cup\,\Emass\,\dot\cup\,\Egad.
\]
The already defined $\Ectrl$ provides control energy.  The set $\Egad$ is the
denominator-sensitive reservoir described in Section~\ref{sec:extra}.  Both are
fixed before the mass batch is selected.  Their reciprocal load satisfies
\begin{equation}\label{eq:base-load}
 \sum_{e\in\Ectrl\cup\Egad}\frac1e<\frac{3}{2b},
\end{equation}
and in fact the control part is below $3/(4b)$ while the reservoir part tends
to zero with the scale.

Remove from the product pool in \eqref{eq:pair-mass-half} the finite obstruction
set $T$, the control products, and the reservoir products.  The forbidden load
is strictly smaller than the surplus in \eqref{eq:pair-mass-half} at large
scale.  Every remaining term is also smaller than $3/(2b)$ because its
semiprime denominator is at least $2^{2k_0}$.

\begin{lemma}[Greedy reciprocal window]\label{lem:greedy-window}
Let $(x_i)_{i\in I}$ be a finite family of positive reals.  Suppose
$\sum_i x_i\ge t$ and $x_i<g$ for every $i$.  Then some subset $J\subseteq I$
satisfies
\[
          t\le\sum_{j\in J}x_j<t+g.
\]
\end{lemma}

\begin{proof}
Choose a subset minimal by inclusion among those whose sum is at least $t$.
Removing any selected term drops the sum below $t$; hence the selected sum is
less than $t+x_j<t+g$ for each selected $j$.  The empty case is excluded by
$t>0$.
\end{proof}

Apply the lemma to the residual target
$t=3/(2b)-\sum_{e\in\Ectrl\cup\Egad}1/e$ and gap $g=3/(2b)$.  It produces
$\Emass$ such that
\begin{equation}\label{eq:load-window}
       \frac{3}{2b}\le\sum_{e\in E}\frac1e<\frac3b.
\end{equation}
Every element of $E$ is a distinct product of two distinct primes, avoids $T$,
and divides the common period
\begin{equation}\label{eq:period}
       L=b\prod_{p\in\Bset}p.
\end{equation}

Let $\Lambda=\sum_{e\in E}1/e$ and use the uniform weight
\begin{equation}\label{eq:theta}
       \theta=\frac{1/b}{\Lambda}.
\end{equation}
Then
\begin{equation}\label{eq:theta-window}
       \frac13<\theta\le\frac23,
       \qquad
       \sum_{e\in E}\frac{\theta}{e}=\frac1b.
\end{equation}
The upper load bound and $b\ge3$ also give $\Lambda<1$, which is the no-wrap
condition \eqref{eq:no-wrap}.

Finally the inverse-square load obeys
\begin{equation}\label{eq:square-load}
       \sum_{e\in E}\frac1{e^2}\le S_{\mathrm{load}}\sctrl^2
\end{equation}
for an absolute $S_{\mathrm{load}}\ge1$.  Hence, with
\[
       \sE^2=\sum_{e\in E}\frac{\theta(1-\theta)}{e^2},
\]
there is a constant $K_\sigma\ge1$ such that
\begin{equation}\label{eq:variance-comparison}
 \sqrt{\frac29}\,\sctrl\le\sE\le K_\sigma\sctrl.
\end{equation}
The lower bound uses the control edges and the minimum of
$\theta(1-\theta)$ on $[1/3,2/3]$; the upper bound uses
\eqref{eq:square-load} and $\theta(1-\theta)\le1/4$.

\section{The main Fourier arcs}\label{sec:main}

Choose a cutoff $C\ge3$ and
\begin{equation}\label{eq:N}
       N=\left\lceil\frac C{\sctrl}\right\rceil.
\end{equation}
After increasing $k_0$, the following hold simultaneously:
\begin{equation}\label{eq:window-geometry}
 2N<2^{2k_0},\qquad 2N+1\le L,
 \qquad \frac1{\sE}\le N.
\end{equation}
The first inequality is used by the reservoir labels, the second makes the
signed labels inject into $\Z/L\Z$, and the third provides enough Gaussian
mass.

For $|m|\le N$, let
\[
 T_m=\prod_{e\in E}\bigl((1-\theta)+\theta e^{2\pi i m/e}\bigr)
       e^{-2\pi i m/b}.
\]
The exact mass identity in \eqref{eq:theta-window} cancels the linear term in
the logarithm.  A uniform third-order expansion of the Bernoulli
characteristic function gives
\begin{equation}\label{eq:main-expansion}
 \log T_m=-2\pi^2m^2\sE^2+\delta_m,
 \qquad
 |\delta_m|\le C_3\sum_{e\in E}|m/e|^3.
\end{equation}
The scale inequality $2N<2^{2k_0}$ puts every $m/e$ in the Taylor disk, and the
inverse-square load plus eventual exponential domination makes the remainder
at most $1/10$.  Therefore
\begin{equation}\label{eq:per-main-lower}
        \Re T_m\ge0.8e^{-2\pi^2m^2\sE^2}.
\end{equation}
Conjugate symmetry gives $T_{-m}=\overline{T_m}$, so the main sum is real.  An
elementary finite Gaussian lower bound yields
\begin{equation}\label{eq:main-lower}
 \Re\sum_{|m|\le N}T_m
 \ge\frac{c_3}{\sE},
 \qquad
 c_3=0.8\frac{e^{-\pi^2/2}}2>0.
\end{equation}

For every frequency, the Bernoulli factor satisfies
\[
 \left|(1-\theta)+\theta e^{2\pi it}\right|^2
 =1-4\theta(1-\theta)\sin^2(\pi t).
\]
Using $\theta\in[1/3,2/3]$ and the elementary lower bound for
$\sin^2(\pi t)$ in terms of $\normdist t^2$ gives
\begin{equation}\label{eq:fourier-energy}
       |F(h)|\le e^{-(16/9)\QE(h)}.
\end{equation}
This is the handoff from Fourier factors to the two minor-energy estimates.

\section{The block-minor fiber estimate}\label{sec:block-minor}

Map a frequency $h\pmod L$ to its block assignment
\[
       a(h)=(h\pmod p)_{p\in\Bset}.
\]
The block-minor sector consists of frequencies for which
$a(h)\notin\mathfrak M(C)$.  The full period has the form
$L=b\prod_{p\in\Bset}p$, so fixing all block residues leaves at most $b$
frequencies modulo $L$.  More generally, write
\[
       Q_{\mathrm{extra}}(h)=\QE(h)-\Qctrl(a(h))\ge0.
\]
Then each assignment fiber satisfies
\begin{equation}\label{eq:fiber-tail}
 \sum_{\substack{h\text{ in block sector}\\a(h)=a}}
 e^{-(16/9)Q_{\mathrm{extra}}(h)}\le b.
\end{equation}
This includes the bounded-cardinality argument, because every summand is at
most $1$.

Reindexing by fibers and using \eqref{eq:control-full-energy},
\begin{align}
 \sum_{h\text{ in block sector}}e^{-(16/9)\QE(h)}
 &\le
 b\sum_{a\notin\mathfrak M(C)}e^{-(16/9)\Qctrl(a)}\\
 &\le
 \frac{b\bigl(\eta+C_{\mathrm{tail}}e^{-(16/9)C^2/2}\bigr)}{\sctrl}.
 \label{eq:block-budget}
\end{align}
Together with \eqref{eq:fourier-energy}, this is the full block-minor norm
budget.  The factor $b$ is not hidden in a generic big-$O$: it is the exact
CRT fiber multiplicity caused by the denominator factor in $L$.

\section{Squarefree CRT and reservoir damping}\label{sec:extra}

The remaining minor frequencies have block assignments in
$\mathfrak M(C)$.  Thus each carries an integer label $m$ with
$|m|\le C/\sctrl\le N$ and
\begin{equation}\label{eq:block-agreement}
       h\equiv m\pmod p\qquad(p\in\Bset).
\end{equation}
A genuine main frequency is the residue of that same integer $m$ modulo $L$.
The extra sector consists of the other frequencies in the same block fiber.
There are at most $b-1$ siblings for each label and at most $2N+1$ labels, so
\begin{equation}\label{eq:extra-count}
       |S_{\mathrm{extra}}|\le b(2N+1).
\end{equation}

Let $R$ be the set of prime divisors of $b$.  Since $b$ is squarefree,
congruence modulo every $r\in R$ implies congruence modulo $b$.
The block primes are coprime to $b$, so the Chinese remainder theorem gives:

\begin{lemma}[Squarefree sibling mismatch]\label{lem:sibling-mismatch}
If \eqref{eq:block-agreement} holds but $h\not\equiv m\pmod L$, then there is
$r\mid b$, $r$ prime, such that $h\not\equiv m\pmod r$.
\end{lemma}

\begin{proof}
If $h\equiv m$ modulo every prime divisor of $b$, squarefreeness gives
$h\equiv m\pmod b$.  Together with \eqref{eq:block-agreement} and coprimality
of $b$ with the block-prime product, this gives $h\equiv m\pmod L$, a
contradiction.
\end{proof}

For every $r\mid b$, choose a common set $S$ of $G$ large block primes, all at
scale at least $2^{2k_0}$, and include the reservoir edges
\begin{equation}\label{eq:reservoir}
        \Egad=\{rs:r\mid b\text{ prime},\ s\in S\}.
\end{equation}
The labels obey $2|m|<s$ by \eqref{eq:window-geometry}.  For the mismatch prime
$r$ supplied by Lemma~\ref{lem:sibling-mismatch}, every $s\in S$ satisfies
$h\equiv m\pmod s$ and $h\not\equiv m\pmod r$.  The Bernoulli factor at $rs$
therefore has the uniform bound
\begin{equation}\label{eq:one-gadget}
 \left|(1-\theta)+\theta e^{2\pi ih/(rs)}\right|
 \le\sqrt{1-\frac{8}{9r^2}}
 \le\sqrt{1-\frac{8}{9b^2}}=:\beta_b<1.
\end{equation}
All other factors have norm at most $1$, so
\begin{equation}\label{eq:multi-gadget}
       |F(h)|\le\beta_b^G.
\end{equation}
Choose a target damping $D_{\mathrm{mp}}>0$ and then choose $G$ so that
$\beta_b^G\le D_{\mathrm{mp}}$.  By \eqref{eq:extra-count},
\begin{equation}\label{eq:extra-budget}
 \sum_{h\in S_{\mathrm{extra}}}|F(h)|
 \le b(2N+1)D_{\mathrm{mp}}.
\end{equation}
Condition $\mathbf D$ guarantees the existence of $G$ distinct primes in the
chosen high block.  The proof has separated the two mechanisms: the CRT lemma
chooses the mismatch prime, while the reservoir lemma aggregates a uniformly
bounded product of contractions.

\section{Terminal parameter compatibility and proof of the structural theorem}
\label{sec:terminal}

The construction contains many eventual inequalities, but no circular choice
is needed.  This section records the exact hierarchy.

Let $c_3$ be the main constant in \eqref{eq:main-lower}.  Let
$S_{\mathrm{load}}$ be the inverse-square constant in
\eqref{eq:square-load}, and choose $K_\sigma\ge1$ with
$S_{\mathrm{load}}\le4K_\sigma^2$, so that the upper variance comparison in
\eqref{eq:variance-comparison} holds.

\begin{enumerate}[label=\textbf{Step \arabic*.}]
\item Fix the block-minor tolerance
\begin{equation}\label{eq:eta-choice}
       \eta=\frac{c_3}{4K_\sigma b}.
\end{equation}
Theorem~\ref{thm:global-partition} then returns a fixed
$C_{\mathrm{tail}}$ and a base-scale threshold.

\item Choose $C\ge3$ so large that
\begin{equation}\label{eq:gaussian-quarter}
 bC_{\mathrm{tail}}e^{-(16/9)C^2/2}
       <\frac{c_3}{4K_\sigma}.
\end{equation}
This choice is made before $k_0$.

\item Put
\begin{equation}\label{eq:Dmp}
 D_{\mathrm{mp}}
 =\frac{c_3}{4K_\sigma b(2C+3)}.
\end{equation}
Since $\beta_b<1$, choose $G\ge1$ such that $\beta_b^G\le D_{\mathrm{mp}}$.

\item Choose the bottom scale $k_0$ after all previous constants.  It must
simultaneously dominate the thresholds for:
\begin{enumerate}[label=(\alph*)]
\item the global-control partition;
\item the dyadic cardinality needed for $G$ reservoir primes;
\item the reciprocal-window condition $\mathbf M$ and the pair-pool mass;
\item the small control and reservoir base load;
\item avoidance of $T$ and separation from the prime divisors of $b$;
\item the lower bound \eqref{eq:sctrl-lower};
\item the main-window inequalities \eqref{eq:window-geometry};
\item the Taylor remainder in \eqref{eq:main-expansion};
\item the inverse-square comparison \eqref{eq:square-load}.
\end{enumerate}
Every item is eventual in $k_0$ after $b,T,C,G$ are fixed, so their finite
maximum is admissible.
\end{enumerate}

The single terminal budget is
\begin{equation}\label{eq:terminal-budget}
 b\eta+bC_{\mathrm{tail}}e^{-(16/9)C^2/2}
       +bD_{\mathrm{mp}}(2C+3)
       <\frac{c_3}{K_\sigma}.
\end{equation}
The first and third terms are exactly one quarter of $c_3/K_\sigma$, the
second is strictly less than one quarter, and one quarter is left as slack.

Because $N=\lceil C/\sctrl\rceil$,
\begin{equation}\label{eq:N-sigma}
       (2N+1)\sctrl\le2C+3.
\end{equation}
Multiplying \eqref{eq:block-budget} and \eqref{eq:extra-budget} by the
appropriate variance normalization and using
$\sE\le K_\sigma\sctrl$, the terminal budget implies
\begin{equation}\label{eq:minor-beat}
 \sum_{h\in S_m}|F(h)|<\frac{c_3}{\sE}.
\end{equation}
The main sum is real and is at least $c_3/\sE$ by
\eqref{eq:main-lower}.  Lemma~\ref{lem:spectral-selection} therefore selects a
subset $A\subseteq E$ with
\[
       \sum_{e\in A}\frac1e=\frac1b.
\]
The construction of $E$ guarantees that its elements are distinct squarefree
semiprimes and avoid $T$.  This proves
Theorem~\ref{thm:structural-construction}.

\begin{remark}[What the terminal certificate prevents]
The order above prevents four hidden circularities: $C$ is not chosen before
the tail constant is known; $G$ is not chosen before the extra-frequency
budget is fixed; $k_0$ is not used to define a constant that it must later
absorb; and the reciprocal-mass batch is selected only after the control and
reservoir loads have been budgeted.
\end{remark}

\section{The Rosser--Schoenfeld backend}\label{sec:RS}

The released proof does not assume $\mathbf D$ and $\mathbf M$ as unexplained
axioms.  It derives them from two exact statements in
\RS~\cite{RosserSchoenfeld1962}.

\begin{proposition}[\RS, Corollary 3, equation (3.8)]\label{prop:RS-density}
For $x\ge41/2$,
\begin{equation}\label{eq:RS-density}
 \frac{3x}{5\log x}
 <\pi(\lfloor2x\rfloor)-\pi(\lfloor x\rfloor).
\end{equation}
\end{proposition}

\begin{proposition}[\RS, Theorem 5, equations (3.17)--(3.18)]
\label{prop:RS-Mertens}
There is one real constant $B$ such that
\begin{align}
 x>1&\Longrightarrow
 \log\log x+B-\frac1{2(\log x)^2}
 <\sum_{p\le\lfloor x\rfloor}\frac1p,
 \label{eq:RS-lower}\\
 x\ge286&\Longrightarrow
 \sum_{p\le\lfloor x\rfloor}\frac1p
 <\log\log x+B+\frac1{2(\log x)^2}.
 \label{eq:RS-upper}
\end{align}
\end{proposition}

\begin{proposition}[Rosser--Schoenfeld implies $\mathbf D$]
Proposition~\ref{prop:RS-density} implies Definition~\ref{def:D}.
\end{proposition}

\begin{proof}
Set $x=2^k$.  For $k\ge5$, $x\ge32>41/2$, and the powers of $2$ at the two
endpoints are composite.  Thus the prime-counting difference in
\eqref{eq:RS-density} is exactly $|\Pk_k|$.  Finally
\[
       \frac{x}{2\log x}\le\frac{3x}{5\log x}.
\]
\end{proof}

\begin{proposition}[Rosser--Schoenfeld implies $\mathbf M$]
Proposition~\ref{prop:RS-Mertens} implies Definition~\ref{def:M}; one may take
$k_1=20$.
\end{proposition}

\begin{proof}
The inclusive union of blocks $k_0,\ldots,3k_0$ is exactly the primes in
$[2^{k_0},2^{3k_0+1})$.  Apply \eqref{eq:RS-lower} at
$x=2^{3k_0+1}$ and \eqref{eq:RS-upper} at $x=2^{k_0}$, then subtract.  The
constant $B$ cancels.  The logarithmic main term is
\[
 \log\log(2^{3k_0+1})-\log\log(2^{k_0})
 =\log\frac{3k_0+1}{k_0}\ge\log3>1.06.
\]
For $k_0\ge20$, each error term
$1/(2\log^2(2^m))$ is less than $0.003$.  Hence the difference is greater than
$1.06-0.006=1.054>21/20$.
\end{proof}

Combining these propositions with Corollary~\ref{cor:structural-E306} proves
the sufficiency direction of Theorem~\ref{thm:headline}; the necessity was
proved in Section~1.
