\title{Distinct Semiprime Egyptian Fractions for Rationals with Squarefree Denominator}
\author{Yuren Tang}
\date{Revised manuscript workspace, 16 July 2026}

\begin{document}
\maketitle

\begin{abstract}
We prove that a positive rational number is a sum of reciprocals of distinct
squarefree semiprimes if and only if its reduced denominator is squarefree.
The only external analytic theorem used in the paper proof is the prime number
theorem.  By Abel summation we derive the local reciprocal-prime law
\[
  \sum_{2^k\le p<2^{k+1}}\frac1p=\frac1k+o\!\left(\frac1k\right),
\]
which supplies both the eventual cardinality of each dyadic prime block and the
fixed reciprocal mass required across an inclusive long dyadic window.

The arithmetic construction is then separated from this analytic input.  It
combines a finite spectral selection principle with a weighted circle method.
Control semiprimes encode residue assignments across dyadic prime blocks; a
reciprocal-mass pool fixes the Bernoulli load; and denominator-sensitive
reservoir edges damp frequencies invisible to the block residues.  We isolate
the load-bearing estimates: a global control partition obtained from level-set
counting and localization, a block-minor fiber estimate with exact CRT
multiplicity, squarefree-CRT reservoir damping, and a terminal parameter order
that makes the main Gaussian contribution dominate both minor sectors.

The immutable software release \texttt{v0.0.3}, archived under DOI
\texttt{10.5281/zenodo.20767390}, remains the current released machine-checked
evidence.  Its exact nonstandard formal inputs are two explicit estimates of
Rosser and Schoenfeld.  A later frozen refactor reorganizes the downstream proof
around structural analytic interfaces and supplies the mathematical handoff
architecture used here; it is not represented as a released artifact.
\end{abstract}

\section{Result, necessity, and proof architecture}

An integer $n$ is a \emph{squarefree semiprime} if
\[
        n=pq
\]
for two distinct primes $p<q$.  Equivalently, its arithmetic functions satisfy
$\omega(n)=\Omega(n)=2$.

\begin{theorem}[Erd\H{o}s 306 in the present formulation]\label{thm:headline}
Let $q>0$ be rational.  Write $q=a/b$ in lowest terms with $a,b\in\N$ and
$b>0$.  Then $q$ is a finite sum of reciprocals of distinct squarefree
semiprimes if and only if $b$ is squarefree.
\end{theorem}

The necessary direction is immediate but fixes the exact scope.  If $S$ is a
finite set of squarefree integers, then the denominator of
$\sum_{n\in S}1/n$ divides $\lcm(S)$, which is squarefree.  Hence every reduced
denominator arising from squarefree semiprimes is squarefree.

The sufficiency proof has four conceptual layers.

\begin{enumerate}[label=\textbf{Layer \arabic*.}]
\item \emph{One analytic theorem and one local bridge.}  The prime number theorem
      is converted by partial summation into a reciprocal-prime law on each
      dyadic block.
\item \emph{Structural supply.}  The local law gives eventual block cardinality
      and eventual reciprocal mass on the inclusive block window
      $k_0,\ldots,3k_0$.
\item \emph{Finite spectral construction.}  From this supply, for every
      squarefree $b\ge3$ and finite obstruction set $T$, construct a weighted
      family of distinct semiprimes whose main Fourier arcs dominate both minor
      sectors.
\item \emph{Arithmetic closure.}  Convert positivity into an exact reciprocal
      subset, deal with $b=1,2$, iterate the avoiding unit-numerator result, and
      order the final finite set.
\end{enumerate}

This is a principle/handoff organization rather than a traversal of a source
tree.  The PNT bridge is the analytic principle; the structural construction is
the handoff to arithmetic; and the circle method is factored into finite
selection, control, mass, and terminal certificates.

\section{The exact finite statement and avoiding reductions}

For a finite set $T\subset\N$, write $\operatorname{Rep}_T(r)$ when there is a
finite set $S\subset\N$ such that
\begin{enumerate}[label=(\roman*)]
\item every $n\in S$ is a squarefree semiprime;
\item $S\cap T=\varnothing$;
\item $\sum_{n\in S}1/n=r$.
\end{enumerate}
The avoidance parameter is what makes later unions disjoint.

\begin{lemma}[Unit numerator implies arbitrary numerator]\label{lem:numerator-reduction}
Assume that $\operatorname{Rep}_T(1/b)$ holds for every finite $T$.  Then for
every $a\ge1$ and finite $T$ one has $\operatorname{Rep}_T(a/b)$.
\end{lemma}

\begin{proof}
Induct on $a$.  Having obtained $S$ for $a/b$, apply the unit-numerator
hypothesis with obstruction set $T\cup S$ to obtain $U$ for $1/b$.  Then
$S\cap U=\varnothing$, and the reciprocal sums add over $S\cup U$.
\end{proof}

\begin{lemma}[The small denominators]\label{lem:small-b}
If the avoiding unit-numerator assertion holds for squarefree denominators at
least $3$, then it holds for all positive squarefree denominators.
\end{lemma}

\begin{proof}
For $b=2$, successively choose disjoint avoiding representations of $1/3$ and
$1/6$ and use $1/2=1/3+1/6$.  For $b=1$, successively choose disjoint avoiding
representations of $1/2$, $1/3$, and $1/6$ and use
$1=1/2+1/3+1/6$.
\end{proof}

Thus the analytic and combinatorial content is the avoiding statement for
squarefree $b\ge3$.

\section{Prime number theorem and a local reciprocal-prime law}
\label{sec:PNT}

Let $\pi(x)$ denote the number of primes at most $x$.  We use the prime number
theorem in the standard form
\begin{equation}\label{eq:PNT}
        \pi(x)\sim\frac{x}{\log x}
        \qquad(x\to\infty).
\end{equation}
A directly checkable modern statement appears at the beginning of
Soundararajan's lecture notes~\cite{Soundararajan2006}; a standard monograph
locator is Ingham, Theorem~23~\cite{Ingham1990}.  Historically, independent
proofs were given in 1896, including Hadamard's~\cite{Hadamard1896}.

For $x>1$, put
\[
        A(x)=\sum_{x\le p<2x}\frac1p.
\]

\begin{proposition}[Local multiplicative-block law]\label{prop:local-law}
The prime number theorem implies
\begin{equation}\label{eq:local-law-x}
        A(x)=\frac{\log2}{\log x}
              +o\!\left(\frac1{\log x}\right).
\end{equation}
Consequently, for
\[
  \Pk_k=\{p\text{ prime}:2^k\le p<2^{k+1}\},
  \qquad
  A_k=\sum_{p\in\Pk_k}\frac1p,
\]
one has
\begin{equation}\label{eq:local-law-k}
        A_k=\frac1k+o\!\left(\frac1k\right).
\end{proposition}

\begin{proof}
First use the interval $(x,2x]$.  Abel summation for the step function $\pi$
and $f(t)=1/t$ gives
\begin{equation}\label{eq:abel}
 \sum_{x<p\le2x}\frac1p
 =\frac{\pi(2x)}{2x}-\frac{\pi(x)}x
  +\int_x^{2x}\frac{\pi(t)}{t^2}\,dt.
\end{equation}
Define
\[
 \varepsilon(x)=\sup_{t\ge x}
 \left|\frac{\pi(t)\log t}{t}-1\right|.
\]
By \eqref{eq:PNT}, $\varepsilon(x)\to0$.  Uniformly for $t\in[x,2x]$,
\[
 \pi(t)=\frac{t}{\log t}\bigl(1+O(\varepsilon(x))\bigr).
\]
Hence
\begin{align}
 \int_x^{2x}\frac{\pi(t)}{t^2}\,dt
 &=\int_x^{2x}\frac{dt}{t\log t}
   +O\!\left(\varepsilon(x)
     \int_x^{2x}\frac{dt}{t\log t}\right)\notag\\
 &=\log\frac{\log(2x)}{\log x}
   +o\!\left(\frac1{\log x}\right)\notag\\
 &=\frac{\log2}{\log x}
   +O\!\left(\frac1{\log^2x}\right)
   +o\!\left(\frac1{\log x}\right).
 \label{eq:abel-integral}
\end{align}
The boundary term in \eqref{eq:abel} is
\begin{align*}
 \frac{\pi(2x)}{2x}-\frac{\pi(x)}x
 &=\frac1{\log(2x)}-\frac1{\log x}
   +o\!\left(\frac1{\log x}\right)\\
 &=o\!\left(\frac1{\log x}\right).
\end{align*}
This proves \eqref{eq:local-law-x} for $(x,2x]$.  Changing to $[x,2x)$
alters the sum by at most $1/x+1/(2x)$, which is
$o(1/\log x)$.  In the application $x=2^k$, $k\ge2$, both endpoints are
composite, so the two conventions agree exactly.  Since
$\log(2^k)=k\log2$, equation \eqref{eq:local-law-k} follows.
\end{proof}

\begin{remark}[Why ordinary Mertens is not the local input]
The classical relation
\[
 \sum_{p\le x}\frac1p=\log\log x+B+o(1)
\]
is enough for a fixed power window, because subtraction at $X^a$ and $X$
gives $\log a+o(1)$.  It does not by itself give
\eqref{eq:local-law-x}: on a single dyadic block the main term is only
$\asymp1/\log x$, while subtraction leaves an uncontrolled $o(1)$ error.
The PNT plus the short partial-summation argument above gives the needed local
scale without introducing a second load-bearing analytic theorem.  Mertens'
historical result~\cite{Mertens1874} is therefore background rather than a
premise of the paper proof.
\end{remark}

\section{Eventual structural supply and the abstract construction theorem}
\label{sec:supply}

The construction consumes only eventual supply because the bottom scale is
chosen after every fixed constant and finite obstruction.

\begin{definition}[Eventual dyadic density $\mathbf D_{\mathrm{ev}}$]
\label{def:D}
There exists $K_D$ such that for every $k\ge K_D$,
\begin{equation}\label{eq:D}
        |\Pk_k|\ge \frac{2^k}{2\log(2^k)}.
\end{equation}
\end{definition}

\begin{definition}[Eventual reciprocal window $\mathbf M_{\mathrm{ev}}$]
\label{def:M}
There exists $K_M$ such that for every $k_0\ge K_M$,
\begin{equation}\label{eq:M}
   \sum_{k=k_0}^{3k_0}A_k\ge\frac{21}{20}.
\end{equation}
The upper block index is inclusive, so the corresponding prime interval is
$[2^{k_0},2^{3k_0+1})$.
\end{definition}

\begin{proposition}[The local law supplies both interfaces]
\label{prop:local-to-supply}
Equation \eqref{eq:local-law-k} implies
$\mathbf D_{\mathrm{ev}}$ and $\mathbf M_{\mathrm{ev}}$.
\end{proposition}

\begin{proof}
For $p\in\Pk_k$ one has $1/p\le2^{-k}$, so
\begin{equation}\label{eq:card-from-mass}
        A_k\le\frac{|\Pk_k|}{2^k},
        \qquad |\Pk_k|\ge2^kA_k.
\end{equation}
From $A_k\sim1/k$, eventually $A_k\ge0.9/k$.  Since
$0.9>1/(2\log2)$, equation \eqref{eq:card-from-mass} gives
\eqref{eq:D} for all sufficiently large $k$.

Also, eventually $A_k\ge0.98/k$.  Therefore
\[
 \sum_{k=k_0}^{3k_0}A_k
 \ge0.98\sum_{k=k_0}^{3k_0}\frac1k.
\]
The harmonic window tends to $\log3$, and
$0.98\log3>1.05=21/20$.  Hence \eqref{eq:M} holds for all sufficiently
large $k_0$.
\end{proof}

The two roles are deliberately separated.  Condition
$\mathbf D_{\mathrm{ev}}$ provides enough vertices in individual blocks for
the control system and high-prime reservoirs.  Condition
$\mathbf M_{\mathrm{ev}}$ supplies reciprocal load for a semiprime pool.  A
large cardinality does not by itself give the required reciprocal mass, and an
aggregate mass statement does not by itself populate every control block.

\begin{theorem}[Structural avoiding construction]
\label{thm:structural-construction}
Assume $\mathbf D_{\mathrm{ev}}$ and $\mathbf M_{\mathrm{ev}}$.  Let $b\ge3$
be squarefree and let $T\subset\N$ be finite.  Then
$\operatorname{Rep}_T(1/b)$ holds.
\end{theorem}

\begin{corollary}[Structural form of Erd\H{o}s 306]
\label{cor:structural-E306}
The prime number theorem implies the sufficiency direction of
Theorem~\ref{thm:headline}.
\end{corollary}

\begin{proof}
Use Proposition~\ref{prop:local-law},
Proposition~\ref{prop:local-to-supply}, and
Theorem~\ref{thm:structural-construction}; then apply
Lemmas~\ref{lem:small-b} and~\ref{lem:numerator-reduction}.
\end{proof}

Sections~\ref{sec:spectral}--\ref{sec:terminal} prove
Theorem~\ref{thm:structural-construction}.  Their mathematical content is the
constant-first/scale-last construction already isolated in the frozen
principle/handoff architecture; replacing fixed explicit thresholds by eventual
ones changes only the final threshold chosen for $k_0$.

\section{Finite spectral selection}\label{sec:spectral}

The circle method is used in a finite cyclic group, so the existence mechanism
can be stated without analytic number theory.  Let $E$ be a finite set of
positive integers, let $L$ be a common multiple of $E$, let
$q_0\in\Z/L\Z$, and let $\theta_e\in[0,1]$.  Choose each $e\in E$
independently with probability $\theta_e$.  For $A\subseteq E$, set
\[
  w(A)=\prod_{e\in A}\theta_e
       \prod_{e\in E\setminus A}(1-\theta_e).
\]
The weighted number of subsets satisfying
$\sum_{e\in A}L/e\equiv q_0\pmod L$ is
\[
 W_{L,q_0}(E,\theta)
 =\sum_{\substack{A\subseteq E\\
       \sum_{e\in A}L/e\equiv q_0\ (\mathrm{mod}\ L)}}w(A).
\]
Finite Fourier orthogonality gives
\begin{equation}\label{eq:finite-fourier}
 L W_{L,q_0}(E,\theta)
 =\sum_{h=0}^{L-1}
   \prod_{e\in E}\bigl((1-\theta_e)+\theta_e e^{2\pi i h/e}\bigr)
   e^{-2\pi i hq_0/L}.
\end{equation}

\begin{lemma}[Finite spectral selection]\label{lem:spectral-selection}
Partition the frequencies into $S_M\dot\cup S_m$.  If
\[
 \Re\sum_{h\in S_M}F(h)>\sum_{h\in S_m}|F(h)|
\]
and the main sum is real, where $F(h)$ is the summand in
\eqref{eq:finite-fourier}, then $W_{L,q_0}(E,\theta)>0$ and an admissible
subset exists.
\end{lemma}

\begin{proof}
The real part of the full Fourier sum is positive.  Since all subset weights are
nonnegative, at least one admissible subset has positive weight.
\end{proof}

For the reciprocal problem take $q_0=L/b$.  If every $e$ divides $L$ and
\begin{equation}\label{eq:no-wrap}
        \sum_{e\in E}\frac1e<1,
\end{equation}
then the congruence is equivalent to
$\sum_{e\in A}1/e=1/b$, because both sides lie in $[0,1)$.

\section{Dyadic control geometry}\label{sec:control}

Fix squarefree $b\ge3$ and finite $T$.  Choose a bottom scale $k_0$, a
terminal scale $K$, and prime blocks
\[
        P_k\subseteq\Pk_k\qquad(k_0\le k\le K).
\]
Condition $\mathbf D_{\mathrm{ev}}$ permits all required finite selections once
$k_0$ exceeds its threshold.  Put
\[
        \Bset=\bigcup_{k=k_0}^{K}P_k.
\]
All primes in $\Bset$ are chosen larger than $b$ and every element of $T$.

The control graph has vertices $\Bset$.  It contains sufficiently many pairs
inside each block and selected pairs between consecutive blocks.  The
corresponding semiprime set is
\[
        \Ectrl=\{pq:\{p,q\}\text{ is a control pair}\}.
\]
Unique factorization makes the product map injective on prime pairs, so these
denominators are distinct.

For each $p\in\Bset$, a frequency $h$ determines a residue
$a_p=h\pmod p$.  A general assignment is
\[
        a=(a_p)_{p\in\Bset}\in
        \mathcal A:=\prod_{p\in\Bset}\Z/p\Z.
\]
For a control pair $\{p,q\}$, let $x_{pq}(a)$ be the centered CRT
representative determined by $(a_p,a_q)$.  Define
\begin{equation}\label{eq:Qctrl}
 \Qctrl(a)=\sum_{\{p,q\}\in\mathcal C}
       \left(\frac{x_{pq}(a)}{pq}\right)^2,
 \qquad
 \sctrl^2=\sum_{\{p,q\}\in\mathcal C}\frac1{(pq)^2}.
\end{equation}
For an actual frequency $h$,
\[
   \normdist{h/(pq)}=\frac{|x_{pq}(a(h))|}{pq}.
\]
Hence, if the final edge family contains $\Ectrl$,
\begin{equation}\label{eq:control-full-energy}
       \Qctrl(a(h))\le\QE(h)
       :=\sum_{e\in E}\normdist{h/e}^{2}.
\end{equation}

The graph is chosen so that for all sufficiently large $k_0$,
\begin{align}
 \sctrl&\ge \frac{1}{c_\sigma k_0 2^{k_0}},
       \label{eq:sctrl-lower}\\
 \sum_{e\in\Ectrl}\frac1e&<\frac{3}{4b}.
       \label{eq:control-load}
\end{align}
Here $c_\sigma\ge1$ is absolute.  The first estimate comes from the numerous
pairs in the bottom block; the second from the quadratic growth of the
semiprime denominators.  Neither estimate uses reciprocal-window mass.

The main set in assignment space is
\begin{equation}\label{eq:assignment-main}
 \mathfrak M(C)=
 \left\{a:\exists m\in\Z,
   |m|\le \frac C{\sctrl},\quad
   a_p\equiv m\pmod p\ \text{for all }p\in\Bset\right\}.
\end{equation}

\section{The global control partition}\label{sec:global}

This is the principal combinatorial estimate.  Its input is only the block
system and its quadratic energy.

\begin{theorem}[Global control partition]\label{thm:global-partition}
For every $c>0$ and $\eta>0$, there exist a scale $k_*(c,\eta)$ and a
constant $C_{\mathrm{tail}}>0$ such that, for every admissible block system
with $k_0\ge k_*$ and every $C\ge1$,
\begin{equation}\label{eq:global-partition}
 \sum_{a\notin\mathfrak M(C)}e^{-c\Qctrl(a)}
 \le
 \frac{\eta+C_{\mathrm{tail}}e^{-cC^2/2}}{\sctrl}.
\end{equation}
\end{theorem}

We expose the four handoffs used in its proof.

\subsection{Single-block rigidity and finite encoding}

For each block $P_k$, let $Q_k(a)$ be its local control energy.  There are
forcing thresholds $R_k$ and adjacent-boundary thresholds $\Pi_k$.

\begin{lemma}[Cold-block structure]\label{lem:cold-block}
For sufficiently large $k_0$, if $Q_k(a)<R_k$, then there is a dominant
integer label $\ell_k$ such that all but an energy-controlled exceptional set
of primes in $P_k$ satisfy $a_p\equiv\ell_k\pmod p$.  The number of
possible labels and exceptional sets with energy at most $R$ is bounded by an
exponential charge $e^{\varepsilon R}$ times a polynomial label window.
\end{lemma}

\begin{proof}[Mechanism]
For two residues in one block, closeness of their reciprocal phases forces a
linear congruence modulo the product of the primes.  Centered CRT
representatives are unique in the interval allowed by the energy threshold.
Thus many low-energy pair constraints decode the same integer fingerprint.  If
no label captures most residues, cross-label pairs force quadratic energy.
Each exception then pays a definite charge.  Summing exceptional sets uses
$\sum_A\prod_{j\in A}u_j=\prod_j(1+u_j)$ with exponentially weighted
charges.
\end{proof}

\begin{lemma}[Adjacent-scale mismatch penalty]\label{lem:boundary-penalty}
If consecutive cold blocks have distinct dominant labels, the adjacent control
pairs force energy at least $\Pi_k$ at their boundary.
\end{lemma}

\begin{proof}[Mechanism]
Residues matching $\ell_k$ in $P_k$ and $\ell_{k+1}$ in $P_{k+1}$
determine centered CRT representatives.  If the labels differ, sufficiently
many cross pairs have nonzero representatives of controlled size.  Squaring and
summing gives the boundary floor; exceptional primes cost only their already
charged energy.
\end{proof}

An assignment of total energy at most $R$ is encoded by the hot blocks, the
cold boundaries with label mismatch, dyadic energy shells, one label per cold
segment, and the residual fiber.  Hot blocks and mismatches consume definite
energy; labels propagate along a segment.

\begin{lemma}[Global level-set bound]\label{lem:level-set}
For every $0<\varepsilon<1$ there are constants $A>0$ and $k_\varepsilon$
such that, for $k_0\ge k_\varepsilon$ and $R\ge1$,
\begin{equation}\label{eq:level-set}
 \#\{a\in\mathcal A:\Qctrl(a)\le R\}
 \le e^{A J}e^{8\varepsilon R}
       \left(1+\frac{\sqrt R}{\sctrl}\right),
\end{equation}
where $J$ is the number of blocks.
\end{lemma}

\begin{proof}
Sum the encoding fibers over hot blocks, mismatch boundaries, shells, and
segment labels.  The weighted entropy sums for hot blocks and boundaries each
contribute at most $e^{2\varepsilon R+J}$; shell counting and exception
fibers contribute $e^{4\varepsilon R+A_0J}$.  Once the first cold-segment
label is fixed, propagation determines the rest except across charged
boundaries.  The initial label has size $O(\sqrt R/\sctrl)$ because a
diagonal label $m$ already contributes $m^2\sctrl^2$.
\end{proof}

\subsection{Localization outside the main assignment arc}

Let $\mathcal F_0=\min(R_{k_0},\Pi_{k_0})$ be the global control floor, and
put
\[
 \mathfrak D(C)=\left\{a:\exists m\in\Z,
  a_p\equiv m\pmod p\ \forall p,
  |m|>C/\sctrl,\quad \Qctrl(a)=m^2\sctrl^2\right\}.
\]

\begin{lemma}[Localization dichotomy]\label{lem:localization}
For all sufficiently large $k_0$,
\[
 a\notin\mathfrak M(C)
 \quad\Longrightarrow\quad
 \Qctrl(a)\ge\mathcal F_0\quad\text{or}\qquada\in\mathfrak D(C).
\]
\end{lemma}

\begin{proof}
A hot block gives the first alternative.  If all blocks are cold but a boundary
label changes, Lemma~\ref{lem:boundary-penalty} gives the first alternative.
Otherwise all labels agree and low-energy merger removes exceptions, so every
residue is the reduction of one integer $m$.  If $|m|\le C/\sctrl$, the
assignment is main; otherwise the CRT identity gives
$\Qctrl(a)=m^2\sctrl^2$.
\end{proof}

\subsection{High-energy absorption}

\begin{lemma}[Laplace mass above the floor]\label{lem:floor-laplace}
For every $c>0$ and $\eta>0$, after increasing $k_0$,
\begin{equation}\label{eq:floor-laplace}
 \sum_{\Qctrl(a)\ge\mathcal F_0}e^{-c\Qctrl(a)}
 \le\frac{\eta}{\sctrl}.
\end{equation}
\end{lemma}

\begin{proof}
Choose $\varepsilon$ with $8\varepsilon<c$ and put
$\bar c=(8\varepsilon+c)/2$.  Decompose assignment space into integer
energy shells and apply Lemma~\ref{lem:level-set}.  The factor
$e^{8\varepsilon R}$ is summable against $e^{-\bar cR}$, giving
\[
 \sum_a e^{-\bar c\Qctrl(a)}
 \le e^{AJ}K_0\left(1+\frac1{\sctrl}\right).
\]
On $\Qctrl\ge\mathcal F_0$, extract
$e^{-(c-\bar c)\mathcal F_0}$.  The forcing floor grows fast enough to
absorb $e^{AJ}$ and $K_0$.  Since $\sctrl\le1$ at large scale, the remaining
coefficient is at most $\eta$.
\end{proof}

\subsection{The diagonal Gaussian sector}

\begin{lemma}[Diagonal Gaussian tail]\label{lem:diagonal-tail}
For every $c>0$ there is $C_{\mathrm{tail}}>0$ such that, for all large $k_0$
and $C\ge1$,
\begin{equation}\label{eq:diagonal-tail}
 \sum_{a\in\mathfrak D(C)}e^{-c\Qctrl(a)}
 \le \frac{C_{\mathrm{tail}}e^{-cC^2/2}}{\sctrl}.
\end{equation}
\end{lemma}

\begin{proof}
A diagonal assignment is determined by its integer label, so the sum injects
into
\[
 \sum_{|m|>C/\sctrl}e^{-cm^2\sctrl^2}.
\]
Split the exponent in half.  The window extracts $e^{-cC^2/2}$, while the
remaining integer Gaussian sum is bounded by a constant depending only on $c$
divided by $\sctrl$.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:global-partition}]
Apply Lemma~\ref{lem:localization}; split the off-main sum into the high-floor
and diagonal sectors; then use Lemmas~\ref{lem:floor-laplace} and
\ref{lem:diagonal-tail}.
\end{proof}

The entropy factor $e^{AJ}$ has $A$ fixed before the final base-scale
threshold.  This quantifier order is what permits the forcing floor to absorb
the entropy.

\section{Reciprocal mass, edge families, and weights}\label{sec:mass}

Condition $\mathbf M_{\mathrm{ev}}$ enters here and nowhere in the global
control proof.  Put
\[
 \Pk(k_0)=\bigcup_{k=k_0}^{3k_0}\Pk_k,
 \qquad S_1=\sum_{p\in\Pk(k_0)}\frac1p,
 \qquad S_2=\sum_{p\in\Pk(k_0)}\frac1{p^2}.
\]
For products of two distinct primes,
\begin{equation}\label{eq:pair-mass-id}
 \sum_{\substack{p,q\in\Pk(k_0)\\p<q}}\frac1{pq}
 =\frac{S_1^2-S_2}{2}.
\end{equation}
Condition $\mathbf M_{\mathrm{ev}}$ gives $S_1\ge21/20$ after its threshold,
while
\[
 S_2\le\sum_{n\ge2^{k_0}}\frac1{n^2}
 \le\frac1{2^{k_0}-1}.
\]
After increasing $k_0$,
\begin{equation}\label{eq:pair-mass-half}
 \sum_{p<q}\frac1{pq}\ge\frac12.
\end{equation}

The final family is the disjoint union
\[
        E=\Ectrl\,\dot\cup\,\Emass\,\dot\cup\,\Egad.
\]
The control set supplies energy.  The reservoir $\Egad$ is defined in
Section~\ref{sec:extra}.  Both are fixed before the mass batch.  Their
reciprocal load satisfies
\begin{equation}\label{eq:base-load}
 \sum_{e\in\Ectrl\cup\Egad}\frac1e<\frac{3}{2b},
\end{equation}
and the reservoir part tends to zero with the scale.

Remove from the product pool the finite obstruction $T$, the control products,
and the reservoir products.  Their total reciprocal load is smaller than the
surplus in \eqref{eq:pair-mass-half} at sufficiently large scale.  Every
remaining term is smaller than $3/(2b)$.

\begin{lemma}[Greedy reciprocal window]\label{lem:greedy-window}
Let $(x_i)_{i\in I}$ be a finite family of positive reals.  Suppose
$\sum_i x_i\ge t$ and $x_i<g$ for every $i$.  Then some subset $J\subseteq I$
satisfies
\[
          t\le\sum_{j\in J}x_j<t+g.
\]
\end{lemma}

\begin{proof}
Choose a subset minimal by inclusion among those whose sum is at least $t$.
Removing any selected term drops below $t$, so the selected sum is less than
$t+x_j<t+g$.
\end{proof}

Apply the lemma to
$t=3/(2b)-\sum_{e\in\Ectrl\cup\Egad}1/e$ and gap $g=3/(2b)$.  It produces
$\Emass$ with
\begin{equation}\label{eq:load-window}
       \frac{3}{2b}\le\sum_{e\in E}\frac1e<\frac3b.
\end{equation}
Every element of $E$ is a distinct product of two distinct primes, avoids $T$,
and divides
\begin{equation}\label{eq:period}
       L=b\prod_{p\in\Bset}p.
\end{equation}

Let $\Lambda=\sum_{e\in E}1/e$ and use the uniform weight
\begin{equation}\label{eq:theta}
       \theta=\frac{1/b}{\Lambda}.
\end{equation}
Then
\begin{equation}\label{eq:theta-window}
       \frac13<\theta\le\frac23,
       \qquad
       \sum_{e\in E}\frac{\theta}{e}=\frac1b.
\end{equation}
The upper load and $b\ge3$ give $\Lambda<1$, the no-wrap condition.

The inverse-square load obeys
\begin{equation}\label{eq:square-load}
       \sum_{e\in E}\frac1{e^2}\le S_{\mathrm{load}}\sctrl^2
\end{equation}
for an absolute $S_{\mathrm{load}}\ge1$.  With
\[
       \sE^2=\sum_{e\in E}\frac{\theta(1-\theta)}{e^2},
\]
there is $K_\sigma\ge1$ such that
\begin{equation}\label{eq:variance-comparison}
 \sqrt{\frac29}\,\sctrl\le\sE\le K_\sigma\sctrl.
\end{equation}

\section{The main Fourier arcs}\label{sec:main}

Choose $C\ge3$ and
\begin{equation}\label{eq:N}
       N=\left\lceil\frac C{\sctrl}\right\rceil.
\end{equation}
After increasing $k_0$,
\begin{equation}\label{eq:window-geometry}
 2N<2^{2k_0},\qquad 2N+1\le L,
 \qquad \frac1{\sE}\le N.
\end{equation}
For $|m|\le N$, let
\[
 T_m=\prod_{e\in E}\bigl((1-\theta)+\theta e^{2\pi i m/e}\bigr)
       e^{-2\pi i m/b}.
\]
The exact mass identity cancels the linear logarithmic term.  A uniform
third-order expansion gives
\begin{equation}\label{eq:main-expansion}
 \log T_m=-2\pi^2m^2\sE^2+\delta_m,
 \qquad
 |\delta_m|\le C_3\sum_{e\in E}|m/e|^3.
\end{equation}
The scale inequality puts every $m/e$ in the Taylor disk, and the
inverse-square load makes the remainder at most $1/10$.  Thus
\begin{equation}\label{eq:per-main-lower}
       \Re T_m\ge0.8e^{-2\pi^2m^2\sE^2}.
\end{equation}
Conjugate symmetry makes the main sum real.  A finite Gaussian lower bound gives
\begin{equation}\label{eq:main-lower}
 \Re\sum_{|m|\le N}T_m
 \ge\frac{c_3}{\sE},
 \qquad
 c_3=0.8\frac{e^{-\pi^2/2}}2>0.
\end{equation}

For every frequency,
\[
 \left|(1-\theta)+\theta e^{2\pi it}\right|^2
 =1-4\theta(1-\theta)\sin^2(\pi t).
\]
Using $\theta\in[1/3,2/3]$ and the elementary lower bound for
$\sin^2(\pi t)$ in terms of $\normdist t^2$ gives
\begin{equation}\label{eq:fourier-energy}
       |F(h)|\le e^{-(16/9)\QE(h)}.
\end{equation}

\section{The block-minor fiber estimate}\label{sec:block-minor}

Map $h\pmod L$ to its assignment
$a(h)=(h\pmod p)_{p\in\Bset}$.  The block-minor sector consists of
$a(h)\notin\mathfrak M(C)$.  Since
$L=b\prod_{p\in\Bset}p$, fixing all block residues leaves at most $b$
frequencies modulo $L$.  Write
\[
       Q_{\mathrm{extra}}(h)=\QE(h)-\Qctrl(a(h))\ge0.
\]
Each assignment fiber satisfies
\begin{equation}\label{eq:fiber-tail}
 \sum_{\substack{h\text{ in block sector}\\a(h)=a}}
 e^{-(16/9)Q_{\mathrm{extra}}(h)}\le b.
\end{equation}
Reindexing by fibers and using Theorem~\ref{thm:global-partition},
\begin{align}
 \sum_{h\text{ in block sector}}e^{-(16/9)\QE(h)}
 &\le b\sum_{a\notin\mathfrak M(C)}e^{-(16/9)\Qctrl(a)}\notag\\
 &\le
 \frac{b\bigl(\eta+C_{\mathrm{tail}}e^{-(16/9)C^2/2}\bigr)}{\sctrl}.
 \label{eq:block-budget}
\end{align}
The factor $b$ is the exact CRT fiber multiplicity caused by the denominator
factor in $L$; it is not hidden in big-$O$ notation.

\section{Squarefree CRT and reservoir damping}\label{sec:extra}

The remaining minor frequencies have assignments in $\mathfrak M(C)$, hence an
integer label $m$ with $|m|\le C/\sctrl\le N$ and
\begin{equation}\label{eq:block-agreement}
       h\equiv m\pmod p\qquad(p\in\Bset).
\end{equation}
A genuine main frequency is the residue of $m$ modulo $L$.  The extra sector is
the other frequencies in that block fiber.  There are at most $b-1$ siblings
per label and $2N+1$ labels, so
\begin{equation}\label{eq:extra-count}
       |S_{\mathrm{extra}}|\le b(2N+1).
\end{equation}

Let $R$ be the prime divisors of $b$.

\begin{lemma}[Squarefree sibling mismatch]\label{lem:sibling-mismatch}
If \eqref{eq:block-agreement} holds but $h\not\equiv m\pmod L$, then there is
a prime $r\mid b$ such that $h\not\equiv m\pmod r$.
\end{lemma}

\begin{proof}
If $h\equiv m$ modulo every prime divisor of $b$, squarefreeness gives
$h\equiv m\pmod b$.  Together with block agreement and coprimality, CRT gives
$h\equiv m\pmod L$, a contradiction.
\end{proof}

For every $r\mid b$, choose a common set $S$ of $G$ sufficiently large block
primes, all at scale at least $2^{2k_0}$, and include
\begin{equation}\label{eq:reservoir}
       \Egad=\{rs:r\mid b\text{ prime},\ s\in S\}.
\end{equation}
The labels satisfy $2|m|<s$.  For the mismatch prime and each $s\in S$, the
Bernoulli factor at $rs$ has the uniform bound
\begin{equation}\label{eq:one-gadget}
 \left|(1-\theta)+\theta e^{2\pi ih/(rs)}\right|
 \le\sqrt{1-\frac{8}{9r^2}}
 \le\sqrt{1-\frac{8}{9b^2}}=: \beta_b<1.
\end{equation}
Therefore
\begin{equation}\label{eq:multi-gadget}
       |F(h)|\le\beta_b^G.
\end{equation}
Choose $D_{\mathrm{mp}}>0$ and then $G$ with
$\beta_b^G\le D_{\mathrm{mp}}$.  By \eqref{eq:extra-count},
\begin{equation}\label{eq:extra-budget}
 \sum_{h\in S_{\mathrm{extra}}}|F(h)|
 \le b(2N+1)D_{\mathrm{mp}}.
\end{equation}
Condition $\mathbf D_{\mathrm{ev}}$ supplies the $G$ distinct high primes after
its threshold.  The CRT lemma chooses a mismatch prime; the reservoir product
aggregates uniform contractions.
