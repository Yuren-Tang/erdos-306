\section{Global encoding and localization}\label{sec:global}

Let $J=K+1-k_0$ be the number of blocks and
\[
 \mathcal L(R)=\{a:\Qctrl(a)\le R\}.
\]
An assignment in $\mathcal L(R)$ is encoded by four finite layers:
\begin{enumerate}[label=(\arabic*)]
\item the hot blocks and their local residue vectors, charged by
      $\sum_{k\in H}R_w(k)\le R$;
\item the cold boundaries at which adjacent dominant labels differ, charged by
      $\sum_{k\in B}\Pi_k\le R$;
\item the integer shells $\lfloor Q_k(a)\rfloor$;
\item the dominant labels and exceptional coordinates on maximal cold segments.
\end{enumerate}
The first three layers determine the cold segments.  Proposition
\ref{prop:boundary-penalty} propagates one label along each segment, while
\eqref{eq:exception-count-symbolic} controls the exceptional coordinates.
The first label contributes a factor of order
$1+\sqrt R/\sctrl$; each later segment start has already been charged to a hot
block or a mismatch boundary.

The count must be uniform in the integer written as the segment label.  A
nonwrapped label outside the range allowed by \eqref{eq:label-bound} has empty
fibre.  A wrapped large label may represent the same residues as a small
dominant integer $M$.  Two primes in the agreement class rigidify the lift: two
admissible small integers agreeing modulo both have a difference divisible by
their product but smaller in absolute value.  Replacing the recorded wrapped
label by $M$ therefore injects its fibre into one fixed small-label fibre.  No
unaccounted family of large labels remains.

\begin{proposition}[Entropy-sensitive level-set estimate]\label{prop:level-set}
For every $\gamma>0$, there are constants $A_\gamma>0$ and
$k_\gamma$ such that, for every admissible block system with
$k_0\ge k_\gamma$ and every $R\ge1$,
\begin{equation}\label{eq:level-set}
 \#\mathcal L(R)
 \le e^{A_\gamma J}e^{\gamma R}
 \left(1+\frac{\sqrt R}{\sctrl}\right).
\end{equation}
The constant $A_\gamma$ is fixed before the block system and the bottom scale.
\end{proposition}
\begin{proof}
The four layers form an injective code: full local vectors reconstruct hot
blocks, and on a cold block the segment label together with the exceptional
prime set and exceptional residues reconstructs every coordinate.  Weighted
sums over hot choices and mismatch boundaries are bounded by their energy
charges; shell vectors are bounded by their total integer mass; exception
choices use the finite identity $\sum_A\prod_{j\in A}u_j=\prod_j(1+u_j)$.
After fixing the forcing normalization, taking the blocks sufficiently large
makes the total energy-dependent encoding cost at most $e^{\gamma R}$.  A
bounded amount of local information per block contributes the structural
factor $e^{A_\gamma J}$, and the segment-start labels give the last factor in
\eqref{eq:level-set}.
\end{proof}

Define the global forcing floor
\begin{equation}\label{eq:global-floor}
 \mathcal F_0=\min\{R_w(k_0),\Pi_{k_0}\}.
\end{equation}
After increasing $k_0$, the bottom hot-block scale and bottom boundary scale
are no larger than their counterparts at later blocks.

\begin{lemma}[Localization]\label{lem:localization}
For sufficiently large $k_0$, every $a\notin\mathfrak M(C)$ satisfies either
$\Qctrl(a)\ge\mathcal F_0$ or there is an integer $m$ such that
\begin{equation}\label{eq:diagonal-sector}
 a_p\equiv m\pmod p\quad(p\in\Bset),
 \qquad |m|>C/\sctrl,
 \qquad \Qctrl(a)=m^2\sctrl^2.
\end{equation}
\end{lemma}
\begin{proof}
Suppose $\Qctrl(a)<\mathcal F_0$.  No block is hot and no adjacent cold labels
differ.  The small-forcing refinement eliminates every exception, so all
coordinates have one common label $m$.  The final scale gives $2|m|<pq$ for
every control pair; hence $m$ itself is the centred CRT representative and
\[
 \Qctrl(a)=\sum_{(p,q)\in\Cgraph}(m/pq)^2=m^2\sctrl^2.
\]
If $|m|\le C/\sctrl$, then $a\in\mathfrak M(C)$; otherwise
\eqref{eq:diagonal-sector} holds.
\end{proof}

We now apply the level-set estimate at the Fourier damping exponent introduced
in Section~\ref{sec:main}.  More generally, fix $c>0$ and choose
\begin{equation}\label{eq:gamma-order}
 0<\gamma<c'<c.
\end{equation}

\begin{lemma}[High-floor Laplace absorption]\label{lem:laplace}
For every $c>0$ and $\eta>0$, after increasing $k_0$,
\begin{equation}\label{eq:laplace}
 \sum_{\Qctrl(a)\ge\mathcal F_0}e^{-c\Qctrl(a)}
 \le\frac\eta{\sctrl}.
\end{equation}
\end{lemma}
\begin{proof}
Decompose into unit energy shells and apply Proposition~\ref{prop:level-set} at
$\gamma<c'$.  This gives
\[
 \sum_a e^{-c'\Qctrl(a)}
 \le e^{A_\gamma J}K(c',\gamma)
 \left(1+\frac1{\sctrl}\right).
\]
On $\Qctrl\ge\mathcal F_0$, extract
$e^{-(c-c')\mathcal F_0}$.  The forcing floor grows exponentially in $k_0$,
whereas $J=O(k_0)$ and Proposition~\ref{prop:control-estimates} gives only the
stated growth of $\sctrl^{-1}$.  A final threshold therefore absorbs the
structural block entropy and the variance scale, proving \eqref{eq:laplace}.
\end{proof}

\begin{lemma}[Diagonal Gaussian tail]\label{lem:gaussian-tail}
For every $c>0$ there is $C_{\mathrm{tail}}(c)>0$ such that, eventually,
\begin{equation}\label{eq:gaussian-tail}
 \sum_{\substack{a\text{ satisfying }\eqref{eq:diagonal-sector}}}
 e^{-c\Qctrl(a)}
 \le\frac{C_{\mathrm{tail}}(c)e^{-cC^2/2}}{\sctrl}.
\end{equation}
\end{lemma}
\begin{proof}
A diagonal assignment is determined by its label, so the left side injects into
\[
 \sum_{|m|>C/\sctrl}e^{-cm^2\sctrl^2}.
\]
Split the exponent into two equal parts.  The first is at most $e^{-cC^2/2}$;
the remaining discrete Gaussian is bounded by its central term and a Gaussian
integral, giving $C_{\mathrm{tail}}(c)/\sctrl$.  For example,
$1+6\sqrt2/\sqrt c$ is an admissible finite certificate.
\end{proof}

\begin{theorem}[Global tail estimate]\label{thm:global-partition}
For every $c>0$ and $\eta>0$, there are a threshold $k_*(c,\eta)$ and a constant
$C_{\mathrm{tail}}(c)>0$ such that, for every admissible block system with
$k_0\ge k_*$ and every $C\ge1$,
\begin{equation}\label{eq:global-partition}
 \sum_{a\notin\mathfrak M(C)}e^{-c\Qctrl(a)}
 \le\frac{\eta+C_{\mathrm{tail}}(c)e^{-cC^2/2}}{\sctrl}.
\end{equation}
\end{theorem}
\begin{proof}
Apply Lemma~\ref{lem:localization}; the two alternatives are bounded by
Lemmas~\ref{lem:laplace} and \ref{lem:gaussian-tail}.
\end{proof}

The order is $c,\eta\to\gamma,c',A_\gamma,C_{\mathrm{tail}}\to k_*$ and then
the block system.  The formerly displayed coefficient $8$ in the level-set
exponent is one finite encoding certificate absorbed here into the arbitrary
parameter $\gamma$; the per-block entropy $e^{A_\gamma J}$ is structural.
