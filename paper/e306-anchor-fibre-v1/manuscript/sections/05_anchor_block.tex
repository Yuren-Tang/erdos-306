\section{The anchor block}
\label{sec:anchor-block}

The anchor block has two functions.  At low energy it synchronizes all of its local residue coordinates to one integer label.  Above the synchronization floor it admits a small weighted partition.  We prove these facts first for an abstract prime block and then apply them to $B$.

\subsection{A reciprocal prime code}

Let $Y$ be large and let $Q$ be a set of primes in $[Y,2Y)$ satisfying
\[
 c\frac{Y}{\log Y}\leq N:=|Q|\leq C\frac{Y}{\log Y}.
\]
For an assignment $a=(a_p)_{p\in Q}$ and distinct $p,q\in Q$, let $H_{pq}(a)$ be the unique centred CRT lift with
\[
 H_{pq}\equiv a_p\pmod p,
 \qquad
 H_{pq}\equiv a_q\pmod q,
 \qquad
 -\frac{pq}{2}<H_{pq}\leq\frac{pq}{2}.
\]
Put
\[
 \mathcal Q(a)=\sum_{p<q}\left(\frac{H_{pq}(a)}{pq}\right)^2,
 \qquad
 \sigma_Q^2=\sum_{p<q}\frac1{p^2q^2}\asymp\frac1{Y^2\log^2Y}.
\]
Fix once and for all $\rho\in(0,1/4)$.

\begin{lemma}[Reciprocal dispersion]
\label{lem:reciprocal-dispersion}
Let $q$ be prime with $Y\leq q<4Y$, let $\mathcal F$ be a set of $s\geq16$ primes in $[Y,2Y)$ not containing $q$, and let $d\not\equiv0\pmod q$.  Then
\[
 \sum_{p\in\mathcal F}\dist{d p^{-1}/q}^2
 \geq c_D\frac{s^3}{Y^2}
\]
for an absolute constant $c_D>0$.
\end{lemma}

\begin{proof}
Put $\delta=s/(64Y)$.  If $\dist{dp^{-1}/q}\leq\delta$, choose an integer $\ell$ such that
\[
 v=dp^{-1}-\ell q,
 \qquad
 |v|\leq\delta q\leq s/16.
\]
Then $vp\equiv d\pmod q$.  For each fixed nonzero $v$, this determines one residue class of $p$ modulo $q$, and an interval of length at most $q$ contains at most two integers in such a class.  There are at most $2(s/16)+1$ possible values of $v$, so fewer than $s/2$ primes can satisfy the small-distance condition.  At least $s/2$ summands are therefore larger than $\delta^2$, which proves the claim.
\end{proof}

Call an integer $m$ a \emph{dominant label} for $a$ if
\[
 |m|\leq Y^2/2
\]
and at least $(1-\rho)N$ coordinates satisfy $a_p\equiv m\pmod p$.  A dominant label is unique: two dominant classes intersect in at least two primes, whose product would divide the difference of the labels, while that difference has absolute value at most $Y^2$.

\begin{lemma}[Cross-label energy]
\label{lem:cross-label-energy}
Let $C_m,C_{m'}$ be disjoint label classes with $m\neq m'$ and $|m|,|m'|\leq B_0<Y^2/4$.  Suppose
\[
 |C_m|\geq\max(16,128B_0/Y),
 \qquad
 |C_{m'}|\geq2.
\]
Then
\[
 \sum_{p\in C_m}\sum_{q\in C_{m'}}
 \left(\frac{H_{pq}}{pq}\right)^2
 \geq c_E\frac{|C_m|^3(|C_{m'}|-1)}{Y^2}.
\]
\end{lemma}

\begin{proof}
Put $d=m'-m$.  At most one prime in $C_{m'}$ divides $d$, since two such primes would have product larger than $Y^2$, whereas $0<|d|<Y^2/2$.  Fix any remaining $q\in C_{m'}$.  For $p\in C_m$, write $H_{pq}=m+jp$.  Reduction modulo $q$ gives $jp\equiv d\pmod q$, and therefore
\[
 \dist{dp^{-1}/q}
 \leq\frac{|H_{pq}|}{pq}+\frac{|m|}{pq}.
\]
By \cref{lem:reciprocal-dispersion}, at least half of the left-hand sides exceed $|C_m|/(64Y)$.  The size hypothesis gives
\[
 \frac{|m|}{pq}\leq\frac{B_0}{Y^2}
 \leq\frac{|C_m|}{128Y}.
\]
Hence at least half the $p$ satisfy $|H_{pq}|/(pq)\geq |C_m|/(128Y)$, contributing $\gg |C_m|^3/Y^2$ for this $q$.  Sum over all but at most one $q$.
\end{proof}

\subsection{Low energy forces one exact label}

\begin{proposition}[Nondominant forcing]
\label{prop:nondominant-forcing}
There is $c_w>0$ such that, for all sufficiently large $Y$,
\[
 \mathcal Q(a)<c_w\frac{Y}{\log^3Y}
\]
implies that $a$ has a unique dominant label.
\end{proposition}

\begin{proof}
The case $\mathcal Q(a)=0$ is immediate.  Assume
\[
 0<R:=\mathcal Q(a)<c\frac{Y}{\log^3Y},
\]
where $c>0$ will be fixed at the end.  Put
\[
 B_0=A_\rho\frac{\sqrt R\,Y^2}{N}
\]
with $A_\rho$ a sufficiently large constant, and call an unordered pair $\{p,q\}$ bad if $|H_{pq}|>B_0$.  Since $pq<4Y^2$, every bad pair contributes more than $B_0^2/(16Y^4)$ to $R$.  Hence the number of bad pairs is at most
\[
 \frac{16RY^4}{B_0^2}=\frac{16N^2}{A_\rho^2}.
\]
The sum of their degrees is twice this number, so some base prime $p_0$ has bad degree at most
\[
 \frac{32N}{A_\rho^2}\leq\frac{\rho N}{8},
\]
after choosing $A_\rho^2\geq256/\rho$.

For every nonbad neighbour $q$, the lift $H_{p_0q}$ is an integer label for $a_q$, has absolute value at most $B_0$, and is congruent to the fixed residue $a_{p_0}$ modulo $p_0$.  The interval $[-B_0,B_0]$ therefore contains at most
\[
 M_0\leq\frac{2B_0}{p_0}+2\leq2B_0/Y+2
\]
possible labels.  The assumed bound on $R$, together with $N\gg Y/\log Y$, gives $B_0=o(Y^2)$; decreasing $c$ if necessary, we may suppose $B_0<Y^2/4$.

Suppose that no label class is dominant.  Call a label class small when its size is less than
\[
 s_0=256(B_0/Y+1).
\]
Then $s_0\geq16$ and $s_0\geq128B_0/Y$.

First suppose that the small classes contain at least $\rho N/4$ vertices.  Since there are at most $M_0$ label classes,
\begin{equation}
\label{eq:small-class-count}
 \frac{\rho N}{4}\leq M_0s_0\leq C_\rho(B_0/Y+1)^2.
\end{equation}
Write $u=B_0/Y=A_\rho\sqrt R\,Y/N$.  If $u<1$, the right side of \cref{eq:small-class-count} is bounded in terms of $\rho$ alone, contradicting $N\to\infty$.  If $u\geq1$, then \cref{eq:small-class-count} gives
\[
 N\ll_\rho u^2=A_\rho^2\frac{RY^2}{N^2},
\]
and therefore
\begin{equation}
\label{eq:small-class-R}
 R\gg_\rho\frac{N^3}{Y^2}\gg\frac{Y}{\log^3Y}.
\end{equation}
This contradicts the assumed upper bound when $c$ is smaller than the resulting constant.

We may therefore suppose that the small classes contain fewer than $\rho N/4$ vertices.  Remove $p_0$, its bad neighbours and every vertex in a small class.  For sufficiently large $N$, the number removed is at most
\[
 1+\frac{\rho N}{8}+\frac{\rho N}{4}\leq\frac{\rho N}{2}.
\]
The remaining substantial label classes have total size
\[
 S\geq(1-\rho/2)N.
\]
Write their sizes as $n_1,\ldots,n_t$.  Each $n_i\geq s_0$, and, because no dominant label exists, each $n_i\leq(1-\rho)N$.

Apply \cref{lem:cross-label-energy} to ordered pairs of distinct substantial classes.  Each interclass edge is counted in at most the two opposite orientations, so the factor of two is absorbed into the constant and
\begin{equation}
\label{eq:substantial-energy}
 R\gg\frac1{Y^2}\sum_i n_i^3\sum_{j\neq i}(n_j-1).
\end{equation}
Because every other class has at least $s_0$ vertices,
\[
 t-1\leq\frac{S-n_i}{s_0},
\]
and hence
\[
 \sum_{j\neq i}(n_j-1)
 =(S-n_i)-(t-1)
 \geq(1-1/s_0)(S-n_i)
 \geq\frac12(S-n_i).
\]

If some class, say $n_1$, satisfies $n_1\geq S/2$, then
\[
 S-n_1\geq(1-\rho/2)N-(1-\rho)N=\frac{\rho N}{2},
\]
so its term in \cref{eq:substantial-energy} is $\gg_\rho N^4$.  Otherwise $S-n_i\geq S/2$ for every $i$, and power mean gives
\[
 \sum_i n_i^3\geq\frac{S^3}{t^2}\geq\frac{S^3}{M_0^2}.
\]
In either case,
\begin{equation}
\label{eq:substantial-R}
 R\gg_\rho\frac{N^4}{M_0^2Y^2}.
\end{equation}

If $u=B_0/Y<1$, then $M_0\leq4$, so \cref{eq:substantial-R} gives
\[
 R\gg_\rho\frac{N^4}{Y^2}\gg\frac{Y^2}{\log^4Y}.
\]
This exceeds every fixed multiple of $Y/\log^3Y$ for large $Y$, since their ratio is $Y/\log Y\to\infty$; this is the contradiction in the small-$B_0/Y$ branch.  If $u\geq1$, then $M_0\ll B_0/Y$, and
\[
 M_0^2\ll_\rho\frac{B_0^2}{Y^2}=A_\rho^2\frac{RY^2}{N^2}.
\]
Substitution in \cref{eq:substantial-R} yields
\[
 R^2\gg_\rho\frac{N^6}{Y^4},
 \qquad
 R\gg_\rho\frac{N^3}{Y^2}\gg\frac{Y}{\log^3Y},
\]
again contradicting the upper bound for sufficiently small $c$.  Thus a dominant label exists; uniqueness was noted above.
\end{proof}

The next proposition is the zero-exception step: low energy does not merely yield an almost-global label.

\begin{proposition}[Exact anchor rigidity]
\label{prop:exact-cold-rigidity}
After decreasing $c_w$ if necessary, every assignment satisfying
\[
 \mathcal Q(a)<c_w\frac{Y}{\log^3Y}
\]
is exactly represented by one integer $m$:
\[
 a_p\equiv m\pmod p
 \qquad(p\in Q).
\]
Moreover,
\[
 |m|\ll_\rho\frac{\sqrt{\mathcal Q(a)}}{\sigma_Q}.
\]
\end{proposition}

\begin{proof}
By \cref{prop:nondominant-forcing}, there is a dominant class $C$ with label $m$.  For $p,q\in C$, the centred lift is exactly $m$, so
\[
 \mathcal Q(a)
 \geq m^2\sum_{p<q\in C}\frac1{p^2q^2}
 \gg_\rho m^2\sigma_Q^2.
\]
This proves the label bound.  Let $E_0=Q\setminus C$.  The label bound, the low-energy upper bound and the scale estimates for $\sigma_Q$ and $C$ give
\[
 |m|
 \ll_\rho\frac{\sqrt{\mathcal Q(a)}}{\sigma_Q}
 \ll_{\rho,c_w}\sqrt{\frac{Y}{\log^3Y}}\,Y\log Y
 =O_{\rho,c_w}\!\left(\frac{Y^{3/2}}{\sqrt{\log Y}}\right)
 =o(|C|Y),
\]
since $|C|\asymp Y/\log Y$.  Hence $|m|\leq |C|Y/128$ for sufficiently large $Y$.  Fix $q\in E_0$ and set
\[
 d\equiv a_q-m\pmod q,
 \qquad d\not\equiv0\pmod q.
\]
For $p\in C$, write the centred lift as $H_{pq}=m+jp$.  Reducing modulo $q$ gives $jp\equiv d\pmod q$, and therefore
\[
 \dist{dp^{-1}/q}
 \leq \frac{|H_{pq}|}{pq}+\frac{|m|}{pq}.
\]
By reciprocal dispersion, at least $|C|/2$ of the left-hand sides exceed $|C|/(64Y)$.  Since $p,q\geq Y$ and $|m|\leq |C|Y/128$,
\[
 \frac{|m|}{pq}\leq\frac{|C|}{128Y},
\]
so those $p$ satisfy $|H_{pq}|/(pq)\geq |C|/(128Y)$.  The cross edges incident with this exceptional $q$ consequently contribute at least $c|C|^3/Y^2$.  The edge sets obtained from distinct $q\in E_0$ are disjoint, whence
\[
 |E_0|\,c|C|^3/Y^2\leq\mathcal Q(a).
\]
Since $|C|\gg Y/\log Y$,
\[
 |E_0|\ll\mathcal Q(a)Y^2/N^3<C'c_w.
\]
Choose $c_w$ so that $C'c_w<1$.  The integer $|E_0|$ must then be zero.\end{proof}

\subsection{Fingerprint counting}

\begin{lemma}[Fingerprint rigidity]
\label{lem:fingerprint-rigidity}
Fix $\mathcal F\subset Q$ with $|\mathcal F|=s\geq16$ and fix its residues.  For $q\in Q\setminus\mathcal F$ and a residue $w\pmod q$, put
\[
 t_q(w)=\sum_{p\in\mathcal F}
 \left(\frac{H_{pq}(a_p,w)}{pq}\right)^2.
\]
For a fixed absolute $g>0$, at most one residue $w$ satisfies
\[
 t_q(w)<g\frac{s^3}{Y^2}.
\]
\end{lemma}

\begin{proof}
Let $w\neq w'$ and put $d=w-w'\not\equiv0\pmod q$.  For every $p\in\mathcal F$, the difference of the two centred CRT lifts represents $dp^{-1}/q$ on the circle.  Hence
\[
 \dist{dp^{-1}/q}^2
 \leq2\left(\frac{H_{pq}(a_p,w)}{pq}\right)^2
  +2\left(\frac{H_{pq}(a_p,w')}{pq}\right)^2.
\]
After summing and applying \cref{lem:reciprocal-dispersion},
\[
 c_D\frac{s^3}{Y^2}
 \leq\sum_{p\in\mathcal F}\dist{dp^{-1}/q}^2
 \leq2t_q(w)+2t_q(w').
\]
Choose $g$ with $4g<c_D$.  Then two residues cannot both have energy below $gs^3/Y^2$.
\end{proof}

\begin{proposition}[Fingerprint entropy bound]
\label{prop:fingerprint-entropy}
For every $\varepsilon>0$, all sufficiently large blocks satisfy
\[
 \#\{a:\mathcal Q(a)\leq R\}\leq\exp(\varepsilon R)
\]
uniformly for
\[
 R\geq c_wY/\log^3Y.
\]
\end{proposition}

\begin{proof}
First suppose $R\leq c_*N^4/Y^2$, where $c_*>0$ is small.  Choose once and for all a set
\[
 \mathcal F\subset Q,
 \qquad
 |\mathcal F|=s,
 \qquad
 s=\left\lceil A(RY^2)^{1/4}\right\rceil,
\]
independently of the residue assignment being enumerated.  The lower range gives $s\geq16$, while small $c_*$ gives $s\leq N/2$.  There are at most $(2Y)^s$ assignments on $\mathcal F$.  Once these residues are enumerated, the functions $t_q$ are determined.  For a complete assignment,
\[
 \sum_{q\notin\mathcal F}t_q(a_q)\leq R.
\]
By \cref{lem:fingerprint-rigidity}, these fixed functions force all but at most
\[
 u\leq\frac{RY^2}{gs^3}\ll_A s
\]
vertices to one residue.  Choosing the exceptional vertices and residues gives at most
\[
 \exp\bigl(O((s+u)\log Y)\bigr)
 =\exp\bigl(O((RY^2)^{1/4}\log Y)\bigr)
\]
possibilities.  Dividing this exponent by $R\geq c_wY/\log^3Y$ gives
\[
 O(Y^{-1/4}\log^{13/4}Y)=o(1).
\]

If $R>c_*N^4/Y^2$, the total number of assignments is at most $(2Y)^N$, whose logarithm is $O(Y)$, whereas $R\gg Y^2/\log^4Y$.  This again is at most $\varepsilon R$ for large $Y$.
\end{proof}

\subsection{Application to the anchor block}

Apply the preceding results to $B\subset[Z/2,Z)$, with $Y=Z/2$.  For an anchor assignment $a_B$, put
\[
 Q_B(a_B)=\sum_{q<q'\in B}
 \left(\frac{H_{qq'}(a_B)}{qq'}\right)^2,
 \qquad
 \sigma_{B,0}^2=\sum_{q<q'\in B}\frac1{q^2q'^2}
 \asymp\frac1{Z^2\log^2Z}.
\]
There is a forcing floor
\[
 F_B=c_B\frac{Z}{\log^3Z}
\]
such that every anchor assignment with $Q_B<F_B$ has one exact integer label $m$.  Moreover,
\[
 |m|\ll\frac{\sqrt{F_B}}{\sigma_{B,0}}
 =O\!\left(\frac{Z^{3/2}}{\sqrt{\log Z}}\right)
 =o(Z^2).
\]
Consequently $m$ is the centred lift on every internal anchor pair and
\begin{equation}
\label{eq:top-exact-energy}
 Q_B(a_B)=m^2\sigma_{B,0}^2.
\end{equation}

Define the anchor modulus
\[
 T_B(a_B)=
 \prod_{q<q'\in B}
 \left|(1-\theta)+\theta
 \exp\left(2\pi i\frac{H_{qq'}(a_B)}{qq'}\right)\right|.
\]
By \cref{eq:bernoulli-modulus},
\[
 T_B(a_B)\leq\exp(-\kappa_{\rm mod}Q_B(a_B)).
\]
Fix $\varepsilon<\kappa_{\rm mod}$ in \cref{prop:fingerprint-entropy}.  On the unit shell $j\leq Q_B<j+1$, the number of assignments is at most $\exp(\varepsilon(j+1))$, while each weight is at most $\exp(-\kappa_{\rm mod}j)$.  Hence
\[
 \sum_{Q_B\geq F_B}T_B(a_B)
 \leq e^\varepsilon
 \sum_{j\geq\lfloor F_B\rfloor}
 \exp\bigl(- (\kappa_{\rm mod}-\varepsilon)j\bigr)
 \ll\exp(-cF_B),
\]
which is the geometric-series estimate
\begin{equation}
\label{eq:noncoherent-top-tail}
 \sum_{Q_B\geq F_B}T_B(a_B)
 \ll\exp(-cF_B).
\end{equation}
Below the floor, distinct anchor assignments have distinct labels, because two such labels are $o(Z^2)$ and agreement modulo two anchor primes forces equality.  Hence
\[
 \sum_{Q_B<F_B}T_B(a_B)
 \leq\sum_{m\in\mathbb Z}
 \exp(-\kappa_{\rm mod}m^2\sigma_{B,0}^2)
 \ll\sigma_{B,0}^{-1}
 \ll Z\log Z.
\]

\begin{proposition}[Anchor partition]
\label{prop:top-partition}
The complete anchor weight satisfies
\begin{equation}
\label{eq:P-top}
 P_{\mathrm{anc}}:=\sum_{a_B}T_B(a_B)
 \ll Z\log Z,
\end{equation}
while its noncoherent part satisfies \cref{eq:noncoherent-top-tail}.  Every coherent anchor assignment carries a unique exact integer label and obeys \cref{eq:top-exact-energy}.
\end{proposition}

The anchor theorem supplies the actual anchor weights in the fibre average, controls noncoherent assignments, and controls the large coherent-label tail.  These are disjoint contributions.  With synchronization established, the next section turns to the anchor-cross rows and their decoders.
