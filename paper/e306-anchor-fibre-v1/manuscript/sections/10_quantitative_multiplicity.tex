\section{Quantitative multiplicity, optimization and transfer}
\label{sec:quantitative-multiplicity}

Retain a fixed reduced target $t=a/b\in(0,1)$ with squarefree $b$, a fixed admissible exponent $\gamma$, and a fixed forbidden set.  Write $\mathcal E_X$, $L_X$, $\Lambda_X$, $\theta_X$ and $\sigma_X$ for the parameterized objects above.  Let
\[
 M_X=|\mathcal E_X|,
 \qquad
 \mathcal R_X(t)=\left\{A\subseteq\mathcal E_X:
   \sum_{e\in A}\frac1e=t\right\},
\]
and let $P_X(t)$ be the Bernoulli probability of the exact target event.

\subsection{Sharp fixed-target coefficient from the total variance}

By character orthogonality and no-wrap,
\begin{equation}
\label{eq:quantitative-weighted-count}
 P_X(t)=\sum_{A\in\mathcal R_X(t)}
 \theta_X^{|A|}(1-\theta_X)^{M_X-|A|}.
\end{equation}
The basic asymptotics are
\begin{equation}
\label{eq:quantitative-basic-asymptotics}
\begin{aligned}
 \theta_X&\longrightarrow\alpha_{t,\gamma}
 =\frac{2t}{(\log\gamma)^2},\\
 M_X&=(1+o(1))\frac{Z^2}{2\log^2Z},\\
 \log L_X&=(1+o(1))Z=o(M_X).
\end{aligned}
\end{equation}
The variance is the total actual-family variance \cref{eq:total-variance}; its two leading contributions are separated only afterwards by \cref{cor:variance-regimes}.

\begin{corollary}[Quantitative target coefficient]
\label{cor:quantitative-coefficient}
For all sufficiently large $X$,
\begin{equation}
\label{eq:quantitative-coefficient}
 L_XP_X(t)\geq\frac{c_{t,\gamma}}{\sigma_X}
\end{equation}
for a fixed $c_{t,\gamma}>0$.
\end{corollary}

\begin{proof}
The strict six-sector assembly leaves a fixed positive fraction of the major contribution after all minor totals have been subtracted.
\end{proof}

\begin{theorem}[Sharp fixed-target coefficient]
\label{thm:sharp-target-coefficient}
For fixed $t$, $\gamma$ and finite forbidden set,
\begin{equation}
\label{eq:sharp-target-coefficient}
 L_XP_X(t)\sim\frac1{\sqrt{2\pi}\,\sigma_X}.
\end{equation}
Equivalently, with $\alpha=\alpha_{t,\gamma}$,
\begin{equation}
\label{eq:sharp-target-coefficient-total}
 L_XP_X(t)
 \sim
 \left[
 2\pi\alpha(1-\alpha)
 \left(
  \frac1{2X^2\log^2X}
  +\frac{\tau(b)}{Z\log Z}
 \right)
 \right]^{-1/2}.
\end{equation}
The explicit variance regimes give
\begin{equation}
\label{eq:sharp-target-coefficient-regimes}
 L_XP_X(t)\sim
 \begin{cases}
 \displaystyle
 \frac{X\log X}{\sqrt{\pi\alpha(1-\alpha)}},&\gamma>2,\\[8pt]
 \displaystyle
 \frac{\sqrt{Z\log Z}}
 {\sqrt{2\pi\alpha(1-\alpha)\tau(b)}},&1<\gamma\leq2.
 \end{cases}
\end{equation}
\end{theorem}

\begin{proof}
Fix $C\geq1$ while $X\to\infty$.  With $N=\lfloor C/\sigma_X\rfloor$, the uniform complex expansion \cref{eq:major-local-gaussian,eq:aggregate-cubic-remainder} gives
\[
 \sigma_X\sum_{|m|\leq C/\sigma_X}F(m)
 \longrightarrow\int_{-C}^{C}\exp(-2\pi^2u^2)\,du.
\]
The normalized total-variance tail is $O(\exp(-cC^2))$.  For fixed $C$, every remaining sector is $o(\sigma_X^{-1})$, including the universal prime-only transition.  Hence
\[
 \limsup_{X\to\infty}
 \left|\sigma_XL_XP_X(t)-
 \int_{-C}^{C}\exp(-2\pi^2u^2)\,du\right|
 \ll\exp(-cC^2).
\]
Only now let $C\to\infty$.  Since the Gaussian integral is $1/\sqrt{2\pi}$, \cref{eq:sharp-target-coefficient} follows.  Formula \cref{eq:sharp-target-coefficient-total} is the total variance \cref{eq:total-variance}, and the two explicit forms are \cref{cor:variance-regimes}.  No diagonal choice $C=C(X)$ and no moving-target uniformity is used.
\end{proof}

\subsection{Direct-family entropy and support diversity}

Put
\[
 H(u)=-u\log u-(1-u)\log(1-u)
 \qquad(0<u<1),
\]
and let
\[
 K_X\sim\operatorname{Bin}(M_X,\theta_X),
 \qquad
 w_X=\left\lceil2\sqrt{M_X\log L_X}\right\rceil.
\]
Then $\sqrt{M_X}\ll w_X=o(M_X)$.

\begin{theorem}[Mesoscopic entropy at the direct-family rate]
\label{thm:mesoscopic-entropy}
One has
\begin{equation}
\label{eq:mesoscopic-entropy-lower}
 \#\left\{A\in\mathcal R_X(t):
 \bigl||A|-\theta_XM_X\bigr|\leq w_X\right\}
 \geq\exp\bigl([H(\alpha_{t,\gamma})-o(1)]M_X\bigr).
\end{equation}
The number of all subsets of $\mathcal E_X$ in the same window is at most
\begin{equation}
\label{eq:mesoscopic-entropy-upper}
 \exp\bigl([H(\alpha_{t,\gamma})+o(1)]M_X\bigr).
\end{equation}
Thus the lower bound has the optimal exponential rate within the stated cardinality window.
\end{theorem}

\begin{proof}
Hoeffding's inequality gives
\[
 \Prob(|K_X-\theta_XM_X|>w_X)\leq2L_X^{-8}.
\]
By \cref{eq:sharp-target-coefficient}, together with $\log L_X=o(M_X)$ and $|\log\sigma_X|=o(M_X)$,
\[
 \log P_X(t)=-\log L_X-\log\sigma_X+O(1)=-o(M_X),
 \qquad
 P_X(t)=\exp(-o(M_X)).
\]
The Hoeffding error is therefore $o(P_X(t))$, so a $(1-o(1))$ proportion of the exact target mass lies in the window.  A Bernoulli atom at $k=\theta_XM_X+d$, $|d|\leq w_X$, is at most
\[
 \exp\left(-M_XH(\theta_X)+O(w_X)\right).
\]
Divide the target mass in the window by this bound and use \cref{eq:quantitative-basic-asymptotics}.  The upper bound follows from $\binom{M}{k}\leq\exp(MH(k/M))$ and the $\exp(o(M_X))$ possible cardinalities.
\end{proof}

\begin{corollary}[Exact-cardinality extraction]
\label{cor:exact-cardinality-extraction}
There is an integer
\[
 k_X=\theta_XM_X+O(w_X)
\]
such that
\begin{equation}
\label{eq:exact-cardinality-count}
 \#\{A\in\mathcal R_X(t):|A|=k_X\}
 \geq\exp\bigl([H(\alpha_{t,\gamma})-o(1)]M_X\bigr).
\end{equation}
\end{corollary}

\begin{proof}
Pigeonhole among the $2w_X+1=\exp(o(M_X))$ cardinalities in the window.
\end{proof}

\begin{corollary}[Macroscopic diversity at one cardinality]
\label{cor:hamming-separated-family}
There are constants $\rho,\eta>0$ and, for all sufficiently large $X$, a family $\mathcal F_X\subseteq\mathcal R_X(t)$ such that
\[
 |\mathcal F_X|\geq\exp(\eta M_X),
 \qquad
 |A|=k_X\quad(A\in\mathcal F_X),
\]
and
\[
 |A\mathbin\triangle B|\geq\rho M_X
 \qquad(A\neq B\in\mathcal F_X).
\]
\end{corollary}

\begin{proof}
Choose $\rho>0$ so small that $H(\rho)<H(\alpha_{t,\gamma})$.  A Hamming ball of radius $\rho M_X$ has at most $\exp([H(\rho)+o(1)]M_X)$ members.  Greedy packing inside the exact-cardinality family proves the assertion.
\end{proof}

\begin{corollary}[Balanced signed reciprocal relations]
\label{cor:balanced-signed-relations}
There are constants $c,\eta'>0$ such that, for all sufficiently large $X$, there are at least $\exp(\eta'M_X)$ distinct ordered pairs $(\mathcal P,\mathcal N)$ of disjoint subsets of $\mathcal E_X$ satisfying
\[
 \sum_{e\in\mathcal P}\frac1e
 =\sum_{e\in\mathcal N}\frac1e,
 \qquad
 |\mathcal P|=|\mathcal N|\geq cM_X.
\]
\end{corollary}

\begin{proof}
Fix $A_0\in\mathcal F_X$.  For every other $B\in\mathcal F_X$, take $\mathcal P=A_0\setminus B$ and $\mathcal N=B\setminus A_0$.  Equal cardinality and equal reciprocal target give the two equalities, while Hamming separation gives the linear support size.  The map from $B$ is injective.
\end{proof}

\subsection{Exact optimization over the admissible exponent}

For fixed $t$, the admissible interval \cref{eq:admissible-gamma} makes
\[
 \alpha_{t,\gamma}=\frac{2t}{(\log\gamma)^2}
\]
decrease continuously through the open interval $(t,1)$ as $\gamma$ increases.

\begin{theorem}[Entropy optimization]
\label{thm:entropy-optimization}
The exact optimization of $H(\alpha_{t,\gamma})$ over fixed admissible $\gamma$ is as follows.
\begin{enumerate}[label=(\roman*)]
\item If $0<t<1/2$, the maximum is attained and equals $\log2$.  The unique maximizing exponent is
\begin{equation}
\label{eq:optimal-gamma}
 \gamma=\exp(2\sqrt t),
\end{equation}
for which $\alpha_{t,\gamma}=1/2$.
\item If $t=1/2$, the supremum is $\log2$, approached as $\gamma\uparrow\exp(\sqrt2)$, and it is not attained.
\item If $1/2<t<1$, the supremum is $H(t)$, approached as $\gamma\uparrow\exp(\sqrt2)$, and it is not attained.
\end{enumerate}
\end{theorem}

\begin{proof}
Binary entropy is uniquely maximized at $1/2$, increases on $(0,1/2)$ and decreases on $(1/2,1)$.  When $t<1/2$, the value $1/2$ lies in the accessible interval $(t,1)$ and gives $(\log\gamma)^2=4t$.  When $t\geq1/2$, the accessible interval lies in $(1/2,1)$ or begins at its boundary, so entropy increases as $\alpha$ decreases towards the excluded endpoint $t$.
\end{proof}

The word \emph{maximum} in part~(i) and \emph{supremum} in parts~(ii)--(iii) are not interchangeable: the upper load boundary $\lambda_\gamma=1$ is excluded by no-wrap.

\subsection{Universal denominator-height form}

Put
\begin{equation}
\label{eq:D-height}
 D=Z^2=X^{2\gamma}.
\end{equation}
Every member of the direct family is at most $D$, and \cref{eq:family-size-modulus} becomes
\begin{equation}
\label{eq:M-height}
 M_X=(2+o(1))\frac{D}{(\log D)^2}.
\end{equation}

\begin{corollary}[Direct denominator-height form]
\label{cor:direct-height-form}
For all sufficiently large $D$ along the direct real parametrization $X=D^{1/(2\gamma)}$, the target $t$ has at least
\begin{equation}
\label{eq:direct-height-count}
 \exp\left(\left[2H(\alpha_{t,\gamma})-o(1)\right]
 \frac{D}{(\log D)^2}\right)
\end{equation}
representations by denominators at most $D$.  One cardinality
\begin{equation}
\label{eq:direct-height-cardinality}
 k_D=\left[2\alpha_{t,\gamma}+o(1)\right]
 \frac{D}{(\log D)^2}
\end{equation}
supports the same lower bound, and an exponential equal-cardinality subfamily has symmetric differences of order $D/(\log D)^2$.  The balanced signed relations have both support sizes of that same order.
\end{corollary}

\begin{proof}
Combine \cref{eq:M-height} with \cref{thm:mesoscopic-entropy,cor:exact-cardinality-extraction,cor:hamming-separated-family,cor:balanced-signed-relations}.  A fixed forbidden set affects only the threshold.
\end{proof}

For $0<t<1/2$, the optimizing exponent \cref{eq:optimal-gamma} gives the attained direct rate
\begin{equation}
\label{eq:optimal-direct-height-rate}
 \exp\left(\left[2\log2-o(1)\right]
 \frac{D}{(\log D)^2}\right),
 \qquad
 k_D=\left[1+o(1)\right]\frac{D}{(\log D)^2}.
\end{equation}

\subsection{Quantitative proper refinement branching}

\begin{corollary}[Balanced direct-rate branching of proper refinements]
\label{cor:quantitative-refinement-branching}
Let $A_0$ be any finite squarefree-semiprime representation, and choose a distinguished $a_0\in A_0$.  For all sufficiently large $D$, there are at least
\begin{equation}
\label{eq:optimal-refinement-count}
 \exp\left(\left[2\log2-o_{a_0}(1)\right]
 \frac{D}{(\log D)^2}\right)
\end{equation}
representations $A'$ of height at most $D$ satisfying $A_0\prec A'$.  One total cardinality
\[
 K_{A_0,a_0}+\left[1+o_{a_0}(1)\right]\frac{D}{(\log D)^2},
\]
where $K_{A_0,a_0}$ is fixed, carries the same rate, and an exponential equal-cardinality subfamily is Hamming-separated on the same linear scale.
\end{corollary}

\begin{proof}
For every $a\in A_0\setminus\{a_0\}$, choose once and for all a representation $C_a$ of $1/a$, successively forbidding $A_0$ and the earlier chosen blocks.  The $C_a$ are pairwise disjoint and avoid $A_0$.  Put
\[
 \mathcal U=A_0\cup\bigcup_{a\neq a_0}C_a.
\]
Apply the direct family to $t=1/a_0$.  Since a squarefree semiprime satisfies $a_0\geq6$, one has $t<1/2$; choose the maximizing exponent $\gamma=\exp(2/\sqrt{a_0})$ and forbid $\mathcal U$.  For each direct support $C_{a_0}$, set
\[
 A'=C_{a_0}\sqcup\bigsqcup_{a\neq a_0}C_a.
\]
The simultaneous replacement corollary \cref{cor:simultaneous-proper-replacement} gives $A_0\prec A'$.  The union map is injective, all fixed blocks have bounded height and contribute the fixed cardinality shift
\[
 K_{A_0,a_0}=\sum_{a\neq a_0}|C_a|,
\]
and symmetric differences are exactly those of the varying block $C_{a_0}$.  Now use \cref{eq:optimal-direct-height-rate}; enlarging the threshold ensures that every fixed denominator is at most $D$.
\end{proof}

The same conclusion applies to any common refinement of finitely many representations, since such a refinement is itself finite.

\subsection{Balanced direct quantitative transfer}

\begin{corollary}[Global quantitative transfer]
\label{cor:global-quantitative-transfer}
Let $q>0$ have squarefree reduced denominator, and let $\mathcal T$ be finite.  Then, for all sufficiently large $D$, there are at least
\begin{equation}
\label{eq:global-height-count}
 \exp\left(\left[2\log2-o_q(1)\right]
 \frac{D}{(\log D)^2}\right)
\end{equation}
representations of $q$ by distinct squarefree semiprimes at most $D$, all avoiding $\mathcal T$.  Exponentially many occur at one cardinality
\begin{equation}
\label{eq:global-height-cardinality}
 |\mathcal B_0|+
 \left[1+o_q(1)\right]\frac{D}{(\log D)^2},
\end{equation}
and an exponential equal-cardinality Hamming-separated subfamily transfers as well.
\end{corollary}

\begin{proof}
Write $q=c/d$ in lowest terms.  Choose a prime $p\nmid d$ so large that
\[
 t=\frac1p<\min(q,1/2).
\]
Then the residual $q-t$ is nonnegative and has reduced denominator dividing the squarefree integer $dp$; if positive, choose once and for all a representation $\mathcal B_0$ of it, avoiding $\mathcal T$, and if it is zero put $\mathcal B_0=\varnothing$.

Apply the balanced direct family to $t=1/p$ with $\gamma=\exp(2/\sqrt p)$, forbidding $\mathcal T\cup\mathcal B_0$.  Union with $\mathcal B_0$ is injective, shifts cardinalities by a constant, and preserves symmetric differences.  The direct exponent is the attained balanced value $2\log2$ from \cref{eq:optimal-direct-height-rate}.
\end{proof}

The threshold and the harmless $o_q(1)$ term may depend on the fixed auxiliary prime $p$ chosen in the proof.  Within the constructed direct family, the leading exponent is the attained balanced value $2\log2$; no global optimality among all semiprime representations of height at most $D$ is asserted.

\begin{remark}[Boundary of the quantitative statement]
The coefficient asymptotic is fixed-target and fixed-$\gamma$.  The article does not assert simultaneous uniformity as $t$ or $\gamma$ moves, a joint value--cardinality local limit, an asymptotic at every prescribed cardinality, or resilience under arbitrary sparse deletion.  These are different quantifier structures, not qualifications missing from the theorem above.
\end{remark}
