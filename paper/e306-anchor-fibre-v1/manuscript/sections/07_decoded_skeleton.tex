\section{The decoded skeleton and its minor ranges}
\label{sec:decoded-skeleton}

After \cref{prop:global-fibre-error}, it remains to estimate the decoded frequency $h_m$ attached to each coherent anchor label $m$.  Noncoherent assignments are controlled by the anchor partition.  The lower--lower prime-pair factors retained during fibre compression now provide the principal damping.

\subsection{Two elementary tail estimates}

We shall repeatedly pass from pointwise decay to a sum over integer labels.  The following elementary bounds make those passages explicit.

\begin{lemma}[Gaussian and stretched-exponential tails]
\label{lem:tail-summation}
Fix $c>0$ and a nonnegative integer $d$.
\begin{enumerate}[label=(\roman*)]
\item There are constants $C,c'>0$ such that, for $B\geq1$ and $A\geq B$,
\begin{equation}
\label{eq:gaussian-tail-sum}
 \sum_{m\geq A}(1+m/B)^d\exp(-cm^2/B^2)
 \leq CB(1+A/B)^d\exp(-c'A^2/B^2).
\end{equation}
In particular,
\[
 \sum_{m\geq A}\exp(-cm^2/B^2)
 \ll B\exp(-c'A^2/B^2).
\]
\item There are constants $C,c'>0$ and $A_0$ such that, for $A\geq A_0$,
\begin{equation}
\label{eq:stretched-tail-sum}
 \sum_{m\geq A}m^d\exp\left(-\frac{cm}{(\log m)^2}\right)
 \leq C\exp\left(-\frac{c'A}{(\log A)^2}\right).
\end{equation}
\end{enumerate}
\end{lemma}

\begin{proof}
For~(i), compare the sum with the integral of the eventually decreasing function, then put $u=x/B$.  Repeated integration by parts, or the elementary inequality
\[
 \int_v^\infty(1+u)^d e^{-cu^2}\,du
 \ll(1+v)^d e^{-c'v^2}
 \qquad(v\geq1),
\]
gives \cref{eq:gaussian-tail-sum}.

For~(ii), split $[A,\infty)$ into dyadic intervals $[2^jA,2^{j+1}A)$.  The $j$th interval contributes at most
\[
 (2^{j+1}A)^{d+1}
 \exp\left(-\frac{c2^jA}{(\log(2^{j+1}A))^2}\right).
\]
For sufficiently large $A$, the polynomial factor is absorbed by replacing $c$ with a smaller constant, uniformly in $j$; the remaining dyadic series is dominated by its first term.  A second reduction of the constant absorbs the initial factor $A^{d+1}$ and proves \cref{eq:stretched-tail-sum}.
\end{proof}

\subsection{Prime-only Gaussian transition}

For every fixed $\gamma>1$, consider
\begin{equation}
\label{eq:transition-range}
 T_0<|m|\leq X^2/4.
\end{equation}
By \cref{prop:prime-decoder-identification}, every coordinate of $h_m$ indexed by $P$ equals $m$.  Hence every complete-pair factor has phase $m/(pq)$.  Since $pq\geq X^2$ and $|m|\leq X^2/4$, these phases lie in the linear circle range, and \cref{eq:bernoulli-modulus} gives
\begin{equation}
\label{eq:transition-pointwise}
 |F(h_m)|
 \leq\exp\left(-c m^2
   \sum_{p<q\in P}\frac1{p^2q^2}\right)
 \leq\exp\left(-c\frac{m^2}{X^2\log^2X}\right).
\end{equation}
At the lower endpoint,
\begin{equation}
\label{eq:transition-lower-exponent}
 \frac{T_0^2}{X^2\log^2X}
 =\kappa_0^2\frac{\min(X^2,Z)^2}{X^2\log^2X}
 \longrightarrow\infty.
\end{equation}
Indeed the right side is $\asymp X^2/\log^2X$ when $\gamma\geq2$ and $\asymp X^{2\gamma-2}/\log^2X$ when $1<\gamma<2$.

Apply \cref{lem:tail-summation}(i) with $B=X\log X$ and $A=T_0$.  Since $A/B\to\infty$,
\begin{align}
 \sum_{T_0<|m|\leq X^2/4}|F(h_m)|
 &\ll X\log X\exp\left(-c'\frac{T_0^2}{X^2\log^2X}\right)\notag\\
 &=o(1)=o(\sigma_{\mathcal E}^{-1}).
 \label{eq:transition-total}
\end{align}
This sector uses no target coordinate.  It fills the gap left by the shorter full-coordinate range when $\gamma\leq2$ and remains harmless when $\gamma>2$.

\subsection{Adaptive retained-pair damping}

For an integer $m$ with
\[
 X^2/4<|m|\leq M_{\mathrm{dec}},
\]
put $I_m=[2\sqrt{|m|},3\sqrt{|m|}]$.

\begin{proposition}[Adaptive prime interval]
\label{prop:adaptive-prime-interval}
For every such $m$, the interval $I_m$ lies in $[X,\eta Z)$ for all sufficiently large $X$ and contains
\[
 K_m\gg\frac{\sqrt{|m|}}{\log|m|}
\]
primes.  On the decoded skeleton, the retained complete-pair energy satisfies
\begin{equation}
\label{eq:adaptive-energy}
 Q_{\mathrm{pair}}(m)\gg\frac{|m|}{(\log|m|)^2}.
\end{equation}
\end{proposition}

\begin{proof}
The lower inclusion follows from $2\sqrt{|m|}>X$.  At the upper endpoint,
\[
 3\sqrt{|m|}
 \leq\frac{3\sqrt{XZ}}{\log Z}
 =o_\eta(Z)<\eta Z
\]
for every fixed $(\gamma,\eta)$ with $\gamma>1$ and $\eta\in(0,1)$.  Hence every prime in $I_m$ lies in $P\setminus B$, so every pair formed inside $I_m$ is retained on the decoded skeleton.  Here $y=\sqrt{|m|}\geq X/2\to\infty$ throughout the relevant range, so the fixed-ratio prime-supply estimate applies uniformly and gives the count.  For distinct $p,q\in I_m$, one has $1/9\leq|m|/(pq)\leq1/4$, so each retained pair contributes a fixed positive amount of energy; summing over the pairs proves \cref{eq:adaptive-energy}.
\end{proof}

Consequently,
\begin{equation}
\label{eq:adaptive-pointwise}
 |F(h_m)|\leq\exp\left(-c\frac{|m|}{(\log|m|)^2}\right).
\end{equation}
Apply \cref{lem:tail-summation}(ii), with $d=0$ and $A=X^2/4$.  Since $\log A\asymp\log X$,
\begin{align}
 \sum_{X^2/4<|m|\leq M_{\mathrm{dec}}}|F(h_m)|
 &\ll\exp\left(-c'\frac{X^2}{\log^2X}\right)\notag\\
 &=o(1)=o(\sigma_{\mathcal E}^{-1}).
 \label{eq:adaptive-total}
\end{align}

\subsection{One pairwise-disjoint exhaustive partition}

Fix $C\geq1$, to be chosen before $X$, and put
\begin{equation}
\label{eq:N-major}
 N=\left\lfloor\frac{C}{\sigma_{\mathcal E}}\right\rfloor.
\end{equation}
By the total variance \cref{eq:total-variance}, for every fixed $C$,
\begin{equation}
\label{eq:N-below-full}
 N=o(T_0).
\end{equation}
For $\gamma>2$ this is $X\log X=o(X^2)$; for $1<\gamma\leq2$ it is $\sqrt{Z\log Z}=o(Z)$.  Hence, for sufficiently large $X$,
\[
 N<T_0<\frac{X^2}{4}<M_{\mathrm{dec}}.
\]
The first inequality is \cref{eq:N-below-full}; the second follows from $\kappa_0<1/4$ and $\min(X^2,Z)\leq X^2$; and the last follows from
\[
 \frac{M_{\mathrm{dec}}}{X^2/4}
 =\frac{4Z}{X(\log Z)^2}
 \asymp\frac{X^{\gamma-1}}{(\log Z)^2}
 \longrightarrow\infty.
\]

Apply the decoder/nondecoder split first.  Every nondecoder tuple belongs to the terminal error sector.  Among decoded tuples, a noncoherent anchor assignment belongs to that same sector.  A coherent assignment has one unique integer label $m$ and one decoded frequency $h_m$.  For every fixed $\gamma>1$, the following six sectors are pairwise disjoint and exhaustive:
\begin{center}
\begin{tabular}{@{}c p{5.2cm} p{5.0cm}@{}}
\toprule
Sector & skeleton condition & control \\
\midrule
I & $|m|\leq N$ & positive Taylor major \\
II & $N<|m|\leq T_0$ & total-variance Gaussian tail \\
III & $T_0<|m|\leq X^2/4$ & prime-only pair Gaussian \\
IV & $X^2/4<|m|\leq M_{\mathrm{dec}}$ & adaptive retained pairs \\
V & $|m|>M_{\mathrm{dec}}$ & internal anchor energy \\
VI & noncoherent decoded tuple, or any nondecoder tuple & anchor entropy and fibre error \\
\bottomrule
\end{tabular}
\end{center}

\begin{proposition}[Universal six-sector exhaustion]
\label{prop:six-sector-exhaustion}
For every fixed $\gamma>1$ and $\eta\in(0,1)$, the six displayed sectors are pairwise disjoint and exhaustive over all frequencies modulo $L$.
\end{proposition}

\begin{proof}
The CRT coordinate map is a bijection.  The decoder/nondecoder split is disjoint for each anchor assignment.  A decoded coherent assignment has exactly one label and therefore lies in exactly one of the displayed disjoint label ranges.  These cases exhaust all decoded coherent assignments.
\end{proof}

\subsection{Outer totals}

For $|m|>M_{\mathrm{dec}}$, the internal anchor factor and \cref{eq:top-exact-energy} give
\[
 |F(h_m)|\leq\exp(-\kappa_{\rm mod}m^2\sigma_{B,0}^2).
\]
Apply \cref{lem:tail-summation}(i) with $B=\sigma_{B,0}^{-1}$ and $A=M_{\mathrm{dec}}$.  Since
\[
 M_{\mathrm{dec}}\sigma_{B,0}
 \asymp_\eta\frac{X}{\log^3Z}\longrightarrow\infty,
\]
we obtain
\begin{align}
 \sum_{|m|>M_{\mathrm{dec}}}|F(h_m)|
 &\ll_\eta\sigma_{B,0}^{-1}
 \exp(-c_\eta M_{\mathrm{dec}}^2\sigma_{B,0}^2)\notag\\
 &\ll_\eta Z\log Z\exp\left(-c'_\eta\frac{X^2}{\log^6Z}\right)
 =o(1)=o(\sigma_{\mathcal E}^{-1}).
 \label{eq:anchor-tail-total}
\end{align}

Finally, \cref{eq:noncoherent-top-tail,eq:global-fibre-error} give
\begin{equation}
\label{eq:terminal-error-total}
 \sum_{\mathrm{terminal}}|F|
 \ll_\eta\exp(-c_\eta F_B)
   +P_{\mathrm{anc}}(\exp(\Delta)-1)
 =o(1)=o(\sigma_{\mathcal E}^{-1}).
\end{equation}
